    // C4J research-only selected represented-cell reconstruction and local derivatives.
    pub(crate) fn c4j_cells(&self, scales:&[f64], ne:usize)->Vec<(usize,f64,f64,f64,f64,f64,usize)>{
        let mut best:[Option<(usize,f64,f64,f64,f64,f64,usize)>;3]=[None,None,None];
        for k in 0..self.template.number.len(){
            let n=self.template.number[k];let lo=self.template.support_lo[k];let hi=self.template.support_hi[k];let u=self.template.energy_ref[k];
            if n<=0.||hi<=lo{continue}
            let m=support_mean(lo,hi,n,u);if m<=lo||m>=hi{continue}
            let r=scales[k/ne];let width=hi-lo;let gap=(m-lo).min(hi-m);
            if gap<width*1e-5{continue}
            let fit=positive_reconstruct(lo,hi,m).ok();
            let b=fit.map(|f|f.beta.abs()).unwrap_or(width/(2.*gap));
            let cross=FIT_THRESHOLDS_EV.iter().any(|&e|lo*r<e&&e<hi*r);
            let category=if cross{1}else if b>30.{2}else{0};
            let v=(k,n,lo,hi,m,b,category);
            if best[category].is_none_or(|x|n>x.1){best[category]=Some(v)}
        }best.into_iter().flatten().collect()
    }
    pub(crate) fn c4j_local_cell(&self,k:usize,r:f64,mean_shift:f64,log_scale:f64)->Result<Self,ForwardError>{
        let n=self.template.number[k];let lo=self.template.support_lo[k];let hi=self.template.support_hi[k];
        let u=self.template.energy_ref[k]+n*mean_shift;let r=r*log_scale.exp();
        let state=PhotonState{number:vec![n],energy_ref:vec![u],support_lo:vec![lo],support_hi:vec![hi]};
        let cell=prepare_cell(0,&state,r)?;
        Ok(Self{template:state,nodes:cell.nodes,diagnostics:cell.diagnostics})
    }
    pub(crate) fn c4j_fixed_score(&self,r:f64,mean_scale:f64,a:[f64;3],nh:f64,dt:f64,scheme:u8)->Result<([f64;13],[f64;13],f64,f64),ForwardError>{
        let n=self.template.number[0];let m=self.template.energy_ref[0]/n;
        let var=sum(self.nodes.iter().map(|q|q.number*(q.q-m).powi(2)))/n;
        if var<=0.{return Err(bad("C4J_DELTA_VARIANCE"))}
        let model=HHeModel::controlled_fixture();let cd=model.c_cm_s*dt;
        let species=[Absorber::HI,Absorber::HeI,Absorber::HeII];let provider=AtomicProvider::reference();
        let mut shape=[0.;13];let mut geom=[0.;13];let mut dn=0.;let mut du=0.;
        for q in &self.nodes{
            let e=q.q*r;let x=cd*sum((0..3).map(|j|a[j]*q.sigma[j]));let (ex,phi,psi)=exponential_factors(x);
            let (dp,dpsi)=if x<0.05{let(mut f,mut g,mut p,mut fac)=(0.,0.,1.,1.);for k in 1..=14{fac*=(k+1) as f64;let s=if k%2==0{1.}else{-1.};f+=s*k as f64*p/fac;g+=s*k as f64*p/(fac*(k+2)as f64);p*=x;}(f,g)}else{let d=(ex-phi)/x;(d,(-d-psi)/x)};
            let (sur,mean,ds,dm)=match scheme{0=>{let v=1./(1.+x);(v,v,-v*v,-v*v)},1=>(ex,phi,-ex,dp),_=>(phi,psi,dp,dpsi)};
            let score=(q.q-m)*mean_scale/var;let w=q.number;dn+=w*score;du+=w*score*(q.q-m);
            // Exact Verner smooth branch derivative in ln E; no threshold jump in this fixed-node derivative.
            let dsig:[f64;3]=std::array::from_fn(|i|{
                if q.sigma[i]==0.{return 0.}
                let(e0,ya,p,yw,y0,y1):(f64,f64,f64,f64,f64,f64)=match i{0=>(0.4298,32.88,2.963,0.,0.,0.),1=>(13.61,1.469,3.188,2.039,0.4434,2.136),_=>(1.720,32.88,2.963,0.,0.,0.)};
                let xx=e/e0-y0;let yy=(xx*xx+y1*y1).sqrt();let z=(yy/ya).sqrt();
                let dlog=(2.*(xx-1.)/((xx-1.).powi(2)+yw*yw)+(p/2.-5.5-p*0.5*z/(1.+z))*xx/(yy*yy))*e/e0;
                q.sigma[i]*dlog
            });
            let dx=cd*sum((0..3).map(|i|a[i]*dsig[i]));
            shape[0]+=w*score*sur;shape[1]+=w*score*sur*q.q;
            geom[0]+=w*ds*dx;geom[1]+=w*ds*dx*q.q;
            for i in 0..3{
                let sig=q.sigma[i];let rate=model.c_cm_s*nh*sig*w*mean;let event=cd*a[i]*sig*w*mean;
                let dr=model.c_cm_s*nh*w*(dsig[i]*mean+sig*dm*dx);let devent=cd*a[i]*w*(dsig[i]*mean+sig*dm*dx);
                let excess=q.excess_ev[i];let de=if sig>0.{e}else{0.};
                shape[2+i]+=score*rate;shape[5+i]+=score*rate*excess*model.ev_erg;shape[8+i]+=score*event;shape[11]+=score*event*model.threshold_ev[i];shape[12]+=score*event*excess;
                geom[2+i]+=dr;geom[5+i]+=(dr*excess+rate*de)*model.ev_erg;geom[8+i]+=devent;geom[11]+=devent*model.threshold_ev[i];geom[12]+=devent*excess+event*de;
            }
        }
        let _=(species,provider);
        Ok((shape,geom,dn/n,du/(n*mean_scale)-1.))
    }
    pub(crate) fn c4j_replace_cell(&self,k:usize,local:&Self)->Self{
        let mut v=self.clone();
        v.template.energy_ref[k]=local.template.energy_ref[0];
        v.nodes.retain(|x|x.cell!=k);
        let mut added=local.nodes.clone();for q in &mut added{q.cell=k;}
        // Match original cell-major ordering.
        let i=v.nodes.partition_point(|x|x.cell<k);v.nodes.splice(i..i,added);
        v
    }
