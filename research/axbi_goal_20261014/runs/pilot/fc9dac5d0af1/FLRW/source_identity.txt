//! Cold point coupled driver. Binary64 backward Euler with empirical adaptivity;
//! neither an inherited FT03 certificate nor a continuum proof.
use crate::axbi_background::{
    isotropic_initial_radiation_moments, AxbiBackground, BackgroundFamily, Cosmology,
};
use crate::axbi_rates::AxbiProvider;
use crate::axbi_sources::AxbiSource;
use crate::igm_background::{FlatFlrwBackground, FlatFlrwConfig};
use crate::igm_state::{w_erg_per_h_to_ev, w_ev_per_h_to_erg, IgmGasState};
use crate::igm_thermal::{IgmPhotoInput, IgmPointRhs};
use crate::{Absorber, AtomicProvider, CharacteristicRay, ForwardError, HHeModel};
use std::{
    fs,
    io::{BufWriter, Write},

    time::Instant,
};
fn err(s: &'static str) -> ForwardError {
    ForwardError::InvalidInput(s)
}
fn ioerr(_: std::io::Error) -> ForwardError {
    err("AXBI_DRIVER_IO")
}
fn f(x: f64) -> Result<f64, ForwardError> {
    if x.is_finite() {
        Ok(x)
    } else {
        Err(err("AXBI_DRIVER_NONFINITE"))
    }
}
#[derive(Clone, Debug)]
pub struct DriverConfig {
    pub q_ref: f64,
    pub same_h: bool,
    pub end_zbar: f64,
    pub escape_fraction: f64,
    pub initial_temperature_k: f64,
    pub energy_subdivisions: usize,
    pub n_mu: usize,
    pub background_panels: usize,
    pub local_target: f64,
    pub initial_dln_a: f64,
    pub max_steps: usize,
    pub max_attempts: usize,
    pub max_rhs: usize,
    pub max_wall_s: f64,
    pub spectra_every_accepted: usize,
}
impl DriverConfig {
    /// Exact choices frozen in SCENARIO_LOCK; callers select explicit variants.
    pub fn baseline() -> Self {
        Self {
            q_ref: 0.0,
            same_h: false,
            end_zbar: 4.0,
            escape_fraction: 0.2,
            initial_temperature_k: 2.7255 * 21.0_f64.powi(2)
                / (150.0 * (0.0224_f64 / 0.023).powf(0.4)),
            energy_subdivisions: 16,
            n_mu: 8,
            background_panels: 4096,
            local_target: 1e-5,
            initial_dln_a: 1e-4,
            max_steps: 10000,
            max_attempts: 30000,
            max_rhs: 2000000,
            max_wall_s: 1200.0,
            spectra_every_accepted: 60,
        }
    }
    pub fn read(text: &str) -> Result<Self, ForwardError> {
        let mut c = Self::baseline();
        for line in text
            .lines()
            .filter(|s| !s.trim().is_empty() && !s.starts_with('#'))
        {
            let (k, v) = line.split_once('=').ok_or(err("AXBI_CONFIG_FORMAT"))?;
            macro_rules! number {
                ($field:ident) => {
                    c.$field = v.parse().map_err(|_| err("AXBI_CONFIG_VALUE"))?
                };
            }
            match k.trim() {
                "q_ref" => number!(q_ref),
                "same_h" => {
                    c.same_h = match v.trim() {
                        "true" => true,
                        "false" => false,
                        _ => return Err(err("AXBI_CONFIG_VALUE")),
                    }
                }
                "end_zbar" => number!(end_zbar),
                "escape_fraction" => number!(escape_fraction),
                "initial_temperature_k" => number!(initial_temperature_k),
                "energy_subdivisions" => number!(energy_subdivisions),
                "n_mu" => number!(n_mu),
                "background_panels" => number!(background_panels),
                "local_target" => number!(local_target),
                "initial_dln_a" => number!(initial_dln_a),
                "max_steps" => number!(max_steps),
                "max_attempts" => number!(max_attempts),
                "max_rhs" => number!(max_rhs),
                "max_wall_s" => number!(max_wall_s),
                "spectra_every_accepted" => number!(spectra_every_accepted),
                _ => return Err(err("AXBI_CONFIG_KEY")),
            }
        }
        c.validate()?;
        Ok(c)
    }
    fn validate(&self) -> Result<(), ForwardError> {
        if self.spectra_every_accepted == 0
            || !self.q_ref.is_finite()
            || !self.end_zbar.is_finite()
            || !(4.0..20.0).contains(&self.end_zbar)
            || !self.local_target.is_finite()
            || self.local_target <= 0.0
            || self.local_target >= 2e-4
            || !self.initial_dln_a.is_finite()
            || !(1e-10..=0.005).contains(&self.initial_dln_a)
            || self.max_steps == 0
            || self.max_attempts == 0
            || self.max_rhs == 0
            || !self.max_wall_s.is_finite()
            || self.max_wall_s <= 0.0
        {
            return Err(err("AXBI_CONFIG_DOMAIN"));
        }
        Ok(())
    }
}
#[derive(Clone, Debug, Default)]
pub struct Ledgers {
    pub source_n: f64,
    pub source_u: f64,
    pub absorbed_n: f64,
    pub escaped_u: f64,
    pub boundary_n: f64,
    pub boundary_u: f64,
    pub thermal_work: f64,
    pub photon_work: f64,
    pub cmb_input: f64,
    pub rr_events: [f64; 3],
    pub ci_events: [f64; 3],
    pub photo_events: [f64; 3],
    pub dr_events: f64,
    pub ci_floor_events: [f64; 3],
    pub ce_cap_ev: f64,
    pub excluded_dr_ev: f64,
}
#[derive(Clone, Debug)]
pub struct DriverState {
    pub a: f64,
    pub gas: IgmGasState,
    pub photons: Vec<f64>,
    pub ledgers: Ledgers,
    pub next_dln_a: f64,
    pub accepted: usize,
    pub attempts: usize,
    /// Adaptive full-versus-two-half comparisons, distinct from trial starts.
    pub comparisons: usize,
    pub rejects: usize,
    pub consecutive_rejects: usize,
    pub rhs_evaluations: usize,
    pub max_newton_residual: f64,
    pub last_error: f64,
    pub wall_elapsed_s: f64,
}
#[derive(Clone, Debug)]
pub struct Driver {
    pub config: DriverConfig,
    pub background: AxbiBackground,
    pub source: AxbiSource,
    pub energies: Vec<f64>,
    pub n_h0_cm3: f64,
    pub n_he0_cm3: f64,
    signature: Vec<[f64; 3]>,
    initial_energy: f64,
}
impl Driver {
    pub fn new(config: DriverConfig) -> Result<(Self, DriverState), ForwardError> {
        config.validate()?;
        let h0 = 67.4 * 1e5 / 3.0856775814913673e24;
        let a_ref = 1.0_f64 / 21.0;
        let a_end = 1.0 / (1.0 + config.end_zbar);
        let flat = FlatFlrwBackground::new(FlatFlrwConfig {
            h0_per_s: h0,
            omega_r: 9.2e-5,
            omega_m: 0.315,
            omega_b: 0.0224 / 0.674_f64.powi(2),
            omega_lambda: 1.0 - 0.315 - 9.2e-5,
            helium_mass_fraction: 0.2454,
            tcmb0_k: 2.7255,
            ln_a_min: a_ref.ln(),
            ln_a_max: a_end.ln(),
            parameter_source: "AX01_SCENARIO_LOCK".into(),
        })?;
        let n_h0 = flat.n_h_comoving_cm3();
        let f_he = 0.2454 / (4.0 * (1.0 - 0.2454));
        let n_he0 = n_h0 * f_he;
        let bg = AxbiBackground::new(
            Cosmology {
                h0_per_s: h0,
                omega_m: 0.315,
                omega_r: 9.2e-5,
                omega_lambda: 1.0 - 0.315 - 9.2e-5,
            },
            a_ref,
            a_end,
            0.0,
            config.q_ref,
            if config.same_h {
                BackgroundFamily::SameHDiagnostic
            } else {
                BackgroundFamily::PhysicalDensityFixed
            },
            config.background_panels,
        )?;
        let source = AxbiSource::new(
            n_h0,
            config.escape_fraction,
            config.energy_subdivisions,
            config.n_mu,
        )?;
        let mut energies = vec![13.6];
        for cell in &source.spectral_cells {
            energies.push(cell.upper_ev);
        }
        let provider = AtomicProvider::reference();
        let signature = energies
            .iter()
            .map(|&e| {
                Ok([
                    provider.cross_section(Absorber::HI, e)?,
                    provider.cross_section(Absorber::HeI, e)?,
                    provider.cross_section(Absorber::HeII, e)?,
                ])
            })
            .collect::<Result<Vec<_>, ForwardError>>()?;
        let gas = IgmGasState::from_temperature(
            [2e-4, 0.0, 0.0],
            n_h0 / a_ref.powi(3),
            n_he0 / a_ref.powi(3),
            config.initial_temperature_k,
        )?;
        let photons = vec![0.0; energies.len() * source.angular_cells.len()];
        let state = DriverState {
            a: a_ref,
            gas,
            photons,
            ledgers: Ledgers::default(),
            next_dln_a: config.initial_dln_a,
            accepted: 0,
            attempts: 0,
            comparisons: 0,
            rejects: 0,
            consecutive_rejects: 0,
            rhs_evaluations: 0,
            max_newton_residual: 0.0,
            last_error: 0.0,
            wall_elapsed_s: 0.0,
        };
        let mut driver = Self {
            config,
            background: bg,
            source,
            energies,
            n_h0_cm3: n_h0,
            n_he0_cm3: n_he0,
            signature,
            initial_energy: 0.0,
        };
        driver.initial_energy = driver.total_energy(&state)?;
        Ok((driver, state))
    }
    pub fn photon_inventory(&self, state: &DriverState) -> (f64, f64) {
        let ne = self.energies.len();
        let mut n = 0.0;
        let mut u = 0.0;
        for (i, &v) in state.photons.iter().enumerate() {
            n += v;
            u += v * self.energies[i % ne];
        }
        (n, u)
    }
    fn binding(&self, gas: &IgmGasState) -> f64 {
        let chi = HHeModel::controlled_fixture().threshold_ev;
        let [x, y, z] = gas.fractions;
        chi[0] * x + self.n_he0_cm3 / self.n_h0_cm3 * (chi[1] * y + (chi[1] + chi[2]) * z)
    }
    pub fn total_energy(&self, s: &DriverState) -> Result<f64, ForwardError> {
        let (_, u) = self.photon_inventory(s);
        f(w_erg_per_h_to_ev(s.gas.w_erg_per_h)?
            + self.binding(&s.gas)
            + u
            + s.ledgers.escaped_u
            + s.ledgers.boundary_u
            + s.ledgers.thermal_work
            + s.ledgers.photon_work
            - s.ledgers.source_u
            - s.ledgers.cmb_input)
    }
    pub fn residuals(&self, s: &DriverState) -> Result<(f64, f64), ForwardError> {
        let (n, _) = self.photon_inventory(s);
        Ok((
            f(n + s.ledgers.absorbed_n + s.ledgers.boundary_n - s.ledgers.source_n)?,
            f(self.total_energy(s)? - self.initial_energy)?,
        ))
    }
    fn ray(&self, k: usize, a: f64) -> Result<CharacteristicRay, ForwardError> {
        let mu = self.source.angular_cells[k].mu;
        let initial = CharacteristicRay::new(1.0, [(1.0 - mu * mu).sqrt(), 0.0, mu], 0.0)?;
        initial.pullback(
            &self.background.point_at_scale(1.0 / 21.0)?.geometry()?,
            &self.background.point_at_scale(a)?.geometry()?,
        )
    }
    /// Conservative two-neighbor N/U deposit. Boundary stock has one owner.
    pub fn deposit(
        &self,
        stocks: &mut [f64],
        ray: usize,
        energy: f64,
        count: f64,
    ) -> Result<(f64, f64), ForwardError> {
        if !energy.is_finite()
            || energy <= 0.0
            || !count.is_finite()
            || count < 0.0
            || ray >= self.source.angular_cells.len()
            || stocks.len() != self.energies.len() * self.source.angular_cells.len()
        {
            return Err(err("AXBI_DEPOSIT_DOMAIN"));
        }
        if energy < self.energies[0] {
            return Ok((count, count * energy));
        }
        if energy > *self.energies.last().unwrap() {
            return Err(err("AXBI_ABOVE_SUPPORT"));
        }
        let hi = self.energies.partition_point(|&e| e < energy);
        let offset = ray * self.energies.len();
        if hi == 0 || self.energies[hi] == energy {
            stocks[offset + hi] += count;
        } else {
            let lo = hi - 1;
            let w = (energy - self.energies[lo]) / (self.energies[hi] - self.energies[lo]);
            stocks[offset + lo] += count * (1.0 - w);
            stocks[offset + hi] += count * w;
        }
        Ok((0.0, 0.0))
    }
    pub fn geometry_map(&self, s: &mut DriverState, to: f64) -> Result<(), ForwardError> {
        let g0 = self.background.point_at_scale(s.a)?.geometry()?;
        let g1 = self.background.point_at_scale(to)?.geometry()?;
        let mut stocks = vec![0.0; s.photons.len()];
        let ne = self.energies.len();
        for k in 0..self.source.angular_cells.len() {
            let local = self.ray(k, s.a)?;
            let moved = local.pullback(&g0, &g1)?;
            let ratio = moved.energy_ev / local.energy_ev;
            for i in 0..ne {
                let n = s.photons[k * ne + i];
                if n == 0.0 {
                    continue;
                }
                let energy = self.energies[i] * ratio;
                let (bn, bu) = self.deposit(&mut stocks, k, energy, n)?;
                s.ledgers.boundary_n += bn;
                s.ledgers.boundary_u += bu;
                s.ledgers.photon_work += n * (self.energies[i] - energy);
            }
        }
        let old_w = w_erg_per_h_to_ev(s.gas.w_erg_per_h)?;
        s.gas = s.gas.adiabatic_to(s.a, to)?;
        s.ledgers.thermal_work += old_w - w_erg_per_h_to_ev(s.gas.w_erg_per_h)?;
        s.photons = stocks;
        s.a = to;
        let eos = s
            .gas
            .eos(self.n_h0_cm3 / to.powi(3), self.n_he0_cm3 / to.powi(3))?;
        AxbiProvider.rates(eos.temperature_k)?;
        Ok(())
    }
    fn inject(&self, s: &mut DriverState, dt: f64) -> Result<(), ForwardError> {
        let gref = self.background.point_at_scale(1.0 / 21.0)?.geometry()?;
        let current = self.background.point_at_scale(s.a)?.geometry()?;
        let mut weights = Vec::new();
        for cell in &self.source.angular_cells {
            let ray =
                CharacteristicRay::new(1.0, [(1.0 - cell.mu * cell.mu).sqrt(), 0.0, cell.mu], 0.0)?;
            weights.push(cell.count_fraction * ray.solid_angle_jacobian(&gref, &current)?);
        }
        let sum = weights.iter().sum::<f64>();
        let rate = self.source.at_redshift(1.0 / s.a - 1.0)?.per_h_s;
        for (k, w) in weights.iter().enumerate() {
            for cell in &self.source.spectral_cells {
                let n = rate * dt * cell.photon_fraction * w / sum;
                let (bn, bu) = self.deposit(&mut s.photons, k, cell.energy_ev, n)?;
                s.ledgers.source_n += n;
                s.ledgers.source_u += n * cell.energy_ev;
                s.ledgers.boundary_n += bn;
                s.ledgers.boundary_u += bu;
            }
        }
        Ok(())
    }
    fn evaluate(
        &self,
        old: &DriverState,
        unknown: [f64; 4],
        dt: f64,
        counter: &mut usize,
    ) -> Result<([f64; 4], Vec<f64>, IgmPointRhs), ForwardError> {
        if *counter >= self.config.max_rhs {
            return Err(err("AXBI_RHS_CAP"));
        }
        *counter += 1;
        let gas = IgmGasState::new(
            [unknown[0], unknown[1], unknown[2]],
            w_ev_per_h_to_erg(unknown[3])?,
        )?;
        let nh = self.n_h0_cm3 / old.a.powi(3);
        let nhe = self.n_he0_cm3 / old.a.powi(3);
        let [x, y, z] = gas.fractions;
        let absorbers = [nh * (1.0 - x), nhe * (1.0 - y - z), nhe * y];
        let c = HHeModel::controlled_fixture();
        let ne = self.energies.len();
        let mut stocks = Vec::with_capacity(old.photons.len());
        let mut photo = IgmPhotoInput::default();
        for (j, &nstar) in old.photons.iter().enumerate() {
            let i = j % ne;
            let opacity = (0..3)
                .map(|k| absorbers[k] * self.signature[i][k])
                .sum::<f64>();
            let n = f(nstar / (1.0 + dt * c.c_cm_s * opacity))?;
            stocks.push(n);
            for k in 0..3 {
                let rate = c.c_cm_s * nh * self.signature[i][k] * n;
                photo.gamma_s[k] += rate;
                if rate != 0.0 {
                    if self.energies[i] < c.threshold_ev[k] {
                        return Err(err("AXBI_THRESHOLD_OWNERSHIP"));
                    }
                    photo.heat_erg_per_absorber_s[k] +=
                        rate * (self.energies[i] - c.threshold_ev[k]) * c.ev_erg;
                }
            }
        }
        let point = AxbiProvider.point(&gas, nh, nhe, 0.0, 2.7255 / old.a, photo)?;
        let mut residual = [0.0; 4];
        for i in 0..3 {
            residual[i] = unknown[i] - old.gas.fractions[i] - dt * point.rhs.fraction_dt[i];
        }
        residual[3] = unknown[3]
            - w_erg_per_h_to_ev(old.gas.w_erg_per_h)?
            - dt * point.rhs.w_dt_erg_per_h_s / c.ev_erg;
        for &r in &residual {
            f(r)?;
        }
        Ok((residual, stocks, point.rhs))
    }
    fn norm(&self, residual: [f64; 4], old_w: f64, new_w: f64) -> f64 {
        residual[0]
            .abs()
            .max(residual[1].abs())
            .max(residual[2].abs())
            .max(residual[3].abs() / old_w.abs().max(new_w.abs()).max(1e-6))
    }
    pub fn chemical_stage(
        &self,
        s: &mut DriverState,
        dt: f64,
        counter: &mut usize,
    ) -> Result<(), ForwardError> {
        if !dt.is_finite() || dt <= 0.0 {
            return Err(err("AXBI_STAGE_DT"));
        }
        let old = s.clone();
        let old_w = w_erg_per_h_to_ev(old.gas.w_erg_per_h)?;
        let mut x = [
            old.gas.fractions[0],
            old.gas.fractions[1],
            old.gas.fractions[2],
            old_w,
        ];
        for _ in 0..50 {
            let (residual, stocks, rhs) = self.evaluate(&old, x, dt, counter)?;
            let norm = self.norm(residual, old_w, x[3]);
            if norm <= 1e-11 {
                let nh = self.n_h0_cm3 / old.a.powi(3);
                let c = HHeModel::controlled_fixture();
                let nold = old.photons.iter().sum::<f64>();
                let nnew = stocks.iter().sum::<f64>();
                let events = rhs.photo_events_cm3_s.map(|e| e * dt / nh);
                let event_total = events.iter().sum::<f64>();
                if (nold - nnew - event_total).abs() > 1e-10 * old.ledgers.source_n.max(1.0) {
                    return Err(err("AXBI_ABSORPTION_LEDGER"));
                }
                s.gas = IgmGasState::new([x[0], x[1], x[2]], w_ev_per_h_to_erg(x[3])?)?;
                s.photons = stocks;
                s.ledgers.absorbed_n += event_total;
                for k in 0..3 {
                    s.ledgers.photo_events[k] += events[k];
                    s.ledgers.rr_events[k] += rhs.rr_events_cm3_s[k] * dt / nh;
                    s.ledgers.ci_events[k] += rhs.ci_events_cm3_s[k] * dt / nh;
                    s.ledgers.ci_floor_events[k] += rhs.ci_floor_events_cm3_s[k] * dt / nh;
                }
                s.ledgers.dr_events += rhs.dr_events_cm3_s * dt / nh;
                s.ledgers.ce_cap_ev +=
                    rhs.ce_cap_cooling_erg_cm3_s.iter().sum::<f64>() * dt / (nh * c.ev_erg);
                s.ledgers.excluded_dr_ev +=
                    rhs.excluded_dr_cooling_erg_cm3_s * dt / (nh * c.ev_erg);
                s.ledgers.escaped_u += rhs.escape_erg_cm3_s * dt / (nh * c.ev_erg);
                s.ledgers.cmb_input += rhs.cmb_to_gas_erg_cm3_s * dt / (nh * c.ev_erg);
                s.max_newton_residual = s.max_newton_residual.max(norm);
                return Ok(());
            }
            let mut jac = [[0.0; 4]; 4];
            for j in 0..4 {
                let delta = 1e-7 * x[j].abs().max(if j == 3 { 1e-6 } else { 1e-4 });
                let mut xp = x;
                xp[j] += delta;
                let trial = self.evaluate(&old, xp, dt, counter);
                let (rp, sign) = match trial {
                    Ok((r, _, _)) => (r, 1.0),
                    Err(_) => {
                        xp = x;
                        xp[j] -= delta;
                        let (r, _, _) = self.evaluate(&old, xp, dt, counter)?;
                        (r, -1.0)
                    }
                };
                for i in 0..4 {
                    jac[i][j] = (rp[i] - residual[i]) / (sign * delta);
                }
            }
            let direction = solve4(jac, residual.map(|r| -r))?;
            let mut alpha = 1.0;
            let mut accepted = None;
            for _ in 0..40 {
                let trial = std::array::from_fn(|i| x[i] + alpha * direction[i]);
                if let Ok((r, _, _)) = self.evaluate(&old, trial, dt, counter) {
                    if self.norm(r, old_w, trial[3]) < norm {
                        accepted = Some(trial);
                        break;
                    }
                }
                alpha *= 0.5;
            }
            x = accepted.ok_or(err("AXBI_NEWTON_LINESEARCH"))?;
        }
        Err(err("AXBI_NEWTON_ITERATIONS"))
    }
    fn tentative(
        &self,
        old: &DriverState,
        to: f64,
        counter: &mut usize,
    ) -> Result<DriverState, ForwardError> {
        let mut s = old.clone();
        let mid = (old.a * to).sqrt();
        let dt = self.background.point_at_scale(to)?.time_s
            - self.background.point_at_scale(old.a)?.time_s;
        self.geometry_map(&mut s, mid)?;
        self.inject(&mut s, dt)?;
        self.chemical_stage(&mut s, dt, counter)?;
        self.geometry_map(&mut s, to)?;
        self.validate(&s, 1e-8)?;
        let (n0, u0) = self.residuals(old)?;
        let (n1, u1) = self.residuals(&s)?;
        if (n1 - n0).abs() > 1e-10 * s.ledgers.source_n.max(1.0)
            || (u1 - u0).abs() > 1e-10 * s.ledgers.source_u.max(1.0)
        {
            return Err(err("AXBI_STEP_LEDGER"));
        }
        Ok(s)
    }
    pub fn validate(&self, s: &DriverState, tolerance: f64) -> Result<(), ForwardError> {
        let eos = s
            .gas
            .eos(self.n_h0_cm3 / s.a.powi(3), self.n_he0_cm3 / s.a.powi(3))?;
        AxbiProvider.rates(eos.temperature_k)?;
        for &n in &s.photons {
            if f(n)? < 0.0 {
                return Err(err("AXBI_STOCK_NEGATIVE"));
            }
        }
        let (n, u) = self.residuals(s)?;
        if n.abs() > tolerance * s.ledgers.source_n.max(1.0)
            || u.abs() > tolerance * s.ledgers.source_u.max(1.0)
        {
            return Err(err("AXBI_CUMULATIVE_LEDGER"));
        }
        let moments = self.moments(s)?;
        if moments[1].abs() > 1e-12 * moments[0].max(1e-30) {
            return Err(err("AXBI_PARITY"));
        }
        Ok(())
    }
    pub fn error_metric(
        &self,
        full: &DriverState,
        half: &DriverState,
    ) -> Result<f64, ForwardError> {
        let mut e = 0.0_f64;
        for i in 0..3 {
            e = e.max(
                (full.gas.fractions[i] - half.gas.fractions[i]).abs()
                    / half.gas.fractions[i].abs().max(1e-5),
            );
        }
        let w = w_erg_per_h_to_ev(half.gas.w_erg_per_h)?;
        e = e.max((w_erg_per_h_to_ev(full.gas.w_erg_per_h)? - w).abs() / w.abs().max(1e-6));
        let ne = self.energies.len();
        let (n, u) = self.photon_inventory(half);
        let dn = full
            .photons
            .iter()
            .zip(&half.photons)
            .map(|(a, b)| (a - b).abs())
            .sum::<f64>();
        let du = full
            .photons
            .iter()
            .zip(&half.photons)
            .enumerate()
            .map(|(i, (a, b))| (a - b).abs() * self.energies[i % ne])
            .sum::<f64>();
        f(e.max(dn / n.max(1e-8)).max(du / u.max(1e-8)))
    }
    pub fn adaptive_step(&self, s: &mut DriverState) -> Result<bool, ForwardError> {
        if s.accepted >= self.config.max_steps || s.attempts >= self.config.max_attempts {
            return Err(err("AXBI_STEP_CAP"));
        }
        let endpoint = 1.0 / (1.0 + self.config.end_zbar);
        if s.a == endpoint {
            return Ok(true);
        }
        let to = (s.a * s.next_dln_a.exp()).min(endpoint);
        let mid = (s.a * to).sqrt();
        let mut counter = s.rhs_evaluations;
        let mut trial_starts = s.attempts;
        s.comparisons += 1;
        let result = (|| {
            let mut launch = |old: &DriverState, endpoint: f64| {
                if trial_starts >= self.config.max_attempts {
                    return Err(err("AXBI_TENTATIVE_CAP"));
                }
                trial_starts += 1;
                self.tentative(old, endpoint, &mut counter)
            };
            let full = launch(s, to)?;
            let first = launch(s, mid)?;
            let half = launch(&first, to)?;
            let metric = self.error_metric(&full, &half)?;
            Ok::<_, ForwardError>((half, metric))
        })();
        s.rhs_evaluations = counter;
        s.attempts = trial_starts;
        match result {
            Ok((mut next, e)) if e < self.config.local_target && e < 2e-4 => {
                next.accepted = s.accepted + 1;
                next.attempts = s.attempts;
                next.comparisons = s.comparisons;
                next.rejects = s.rejects;
                next.rhs_evaluations = counter;
                next.consecutive_rejects = 0;
                next.last_error = e;
                next.next_dln_a = if e < self.config.local_target * 0.25 {
                    (s.next_dln_a * 2.0).min(0.005)
                } else {
                    s.next_dln_a
                };
                next.wall_elapsed_s = s.wall_elapsed_s;
                *s = next;
                Ok(true)
            }
            outcome => {
                if let Ok((_, metric)) = &outcome {
                    s.last_error = *metric;
                }
                s.rejects += 1;
                s.consecutive_rejects += 1;
                s.next_dln_a *= 0.5;
                if let Err(e) = outcome {
                    return Err(e);
                }
                Ok(false)
            }
        }
    }
    pub fn moments(&self, s: &DriverState) -> Result<[f64; 3], ForwardError> {
        let mut moments = [0.0; 3];
        let ne = self.energies.len();
        for k in 0..self.source.angular_cells.len() {
            let mu = self.ray(k, s.a)?.direction[2];
            for i in 0..ne {
                let u = s.photons[k * ne + i] * self.energies[i];
                moments[0] += u;
                moments[1] += u * mu;
                moments[2] += u * (3.0 * mu * mu - 1.0) / 2.0;
            }
        }
        Ok(moments)
    }
}
fn solve4(mut a: [[f64; 4]; 4], mut b: [f64; 4]) -> Result<[f64; 4], ForwardError> {
    for k in 0..4 {
        let pivot = (k..4)
            .max_by(|&i, &j| a[i][k].abs().total_cmp(&a[j][k].abs()))
            .unwrap();
        if a[pivot][k].abs() < 1e-30 || !a[pivot][k].is_finite() {
            return Err(err("AXBI_SINGULAR_JACOBIAN"));
        }
        a.swap(k, pivot);
        b.swap(k, pivot);
        for i in k + 1..4 {
            let factor = a[i][k] / a[k][k];
            for j in k..4 {
                a[i][j] -= factor * a[k][j];
            }
            b[i] -= factor * b[k];
        }
    }
    let mut x = [0.0; 4];
    for i in (0..4).rev() {
        x[i] = f((b[i] - (i + 1..4).map(|j| a[i][j] * x[j]).sum::<f64>()) / a[i][i])?;
    }
    Ok(x)
}

fn ledger_values(l: &Ledgers) -> Vec<f64> {
    let mut v = vec![

        l.escaped_u,

        l.thermal_work,

    ];

fn ledger_from(v: &[f64]) -> Result<Ledgers, ForwardError> {
    if v.len() != 24 {
        return Err(err("AXBI_CHECKPOINT_LEDGER"));
    }
    for &x in v {
        f(x)?;
    }
    let array = |i| [v[i], v[i + 1], v[i + 2]];
    Ok(Ledgers {
        source_n: v[0],
        source_u: v[1],
        absorbed_n: v[2],
        escaped_u: v[3],
        boundary_n: v[4],
        boundary_u: v[5],
        thermal_work: v[6],
        photon_work: v[7],
        cmb_input: v[8],
        dr_events: v[9],
        ce_cap_ev: v[10],
        excluded_dr_ev: v[11],
        rr_events: array(12),
        ci_events: array(15),
        photo_events: array(18),
        ci_floor_events: array(21),
    })
}
fn list(v: &[f64]) -> String {
    v.iter()
        .map(|x| format!("{x:.17e}"))
        .collect::<Vec<_>>()
        .join(" ")
}
fn parse_list(s: &str) -> Result<Vec<f64>, ForwardError> {
    s.split_whitespace()
        .map(|t| {
            t.parse::<f64>()
                .map_err(|_| err("AXBI_CHECKPOINT_FORMAT"))
                .and_then(f)
        })
        .collect()
}
impl Driver {
    fn checkpoint_text(&self, s: &DriverState) -> Result<String, ForwardError> {
        let p = self.background.point_at_scale(s.a)?;
        let text = format!(
            "AXBI_CHECKPOINT_V2\n{}\n{}\n{}\n{}\n{}\n",

                s.gas.w_erg_per_h,

                s.rhs_evaluations,
                s.comparisons
            ]

            .collect::<Vec<_>>()
            .join(" "),

        );

    }
    pub fn save_checkpoint(&self, path: &Path, s: &DriverState) -> Result<(), ForwardError> {

    pub fn read_checkpoint(&self, path: &Path) -> Result<DriverState, ForwardError> {
        let text = fs::read_to_string(path).map_err(ioerr)?;
        let lines = text.lines().collect::<Vec<_>>();
        if lines.len() != 6 || lines[0] != "AXBI_CHECKPOINT_V2" {
            return Err(err("AXBI_CHECKPOINT_VERSION"));
        }
        let v = parse_list(lines[1])?;
        let counts = lines[2]
            .split_whitespace()
            .map(|s| s.parse::<usize>().map_err(|_| err("AXBI_CHECKPOINT_COUNT")))
            .collect::<Result<Vec<_>, _>>()?;
        let l = ledger_from(&parse_list(lines[3])?)?;
        let photons = parse_list(lines[4])?;
        let initial = parse_list(lines[5])?;
        if v.len() != 11
            || counts.len() != 6
            || photons.len() != self.energies.len() * self.source.angular_cells.len()
            || initial != vec![self.initial_energy]
        {
            return Err(err("AXBI_CHECKPOINT_SHAPE_OR_INITIAL"));
        }
        let p = self.background.point_at_scale(v[0])?;
        if p.time_s != v[1] || p.b != v[2] || !(1e-10..=0.005).contains(&v[7]) || v[10] < 0.0 {
            return Err(err("AXBI_CHECKPOINT_CLOCK_OR_STEP"));
        }
        let state = DriverState {
            a: v[0],
            gas: IgmGasState::new([v[3], v[4], v[5]], v[6])?,
            photons,
            ledgers: l,
            next_dln_a: v[7],
            max_newton_residual: v[8],
            last_error: v[9],
            wall_elapsed_s: v[10],
            accepted: counts[0],
            attempts: counts[1],
            rejects: counts[2],
            consecutive_rejects: counts[3],
            rhs_evaluations: counts[4],
            comparisons: counts[5],
        };
        self.validate(&state, 1e-8)?;
        Ok(state)
    }
    pub fn output_row(
        &self,
        s: &DriverState,
        case: &str,
        history: &mut impl Write,
        abundances: &mut impl Write,
        spectra: &mut impl Write,
        ledgers: &mut impl Write,
        feedback: &mut impl Write,
    ) -> Result<(), ForwardError> {
        let p = self.background.point_at_scale(s.a)?;
        let nh = self.n_h0_cm3 / s.a.powi(3);
        let nhe = self.n_he0_cm3 / s.a.powi(3);
        let eos = s.gas.eos(nh, nhe)?;
        let [x, y, z] = s.gas.fractions;
        let m = self.moments(s)?;
        let (n, u) = self.photon_inventory(s);
        let c = HHeModel::controlled_fixture();
        let ne = self.energies.len();
        let mut gamma = [0.0; 3];
        for (i, &stock) in s.photons.iter().enumerate() {
            for (k, g) in gamma.iter_mut().enumerate() {
                *g += c.c_cm_s * nh * self.signature[i % ne][k] * stock;
            }
        }
        writeln!(
            history,
            "{case},{},{},{},{},{},{},{},{},{x},{y},{z},{nh},{},{},{},{},{},{n},{u},{},{},{},{},{}",
            self.config.q_ref,
            p.sigma,
            p.time_s,
            s.a,
            1.0 / s.a - 1.0,
            p.b,
            p.h_per_s,
            p.s_per_s,
            eos.electron_density_cm3,
            eos.temperature_k,
            gamma[0],
            gamma[1],
            gamma[2],
            m[0],
            m[1],
            m[2],
            1.0 - x,
            1.0 - y - z
        )
        .map_err(ioerr)?;
        let nb = nh + 4.0 * nhe;
        for (id, a, z_nuc, charge, density, fraction) in [
            ("H_I", 1, 1, 0, nh * (1.0 - x), 1.0 - x),
            ("H_II", 1, 1, 1, nh * x, x),
            ("He_I", 4, 2, 0, nhe * (1.0 - y - z), 1.0 - y - z),
            ("He_II", 4, 2, 1, nhe * y, y),
            ("He_III", 4, 2, 2, nhe * z, z),
        ] {
            writeln!(abundances,"{case},{},{},{},{},{id},{a},{z_nuc},{charge},{density},{fraction},{},,BASELINE_H_HE_CHARGE_AND_BARYON_V1",p.time_s,1.0/s.a-1.0,eos.temperature_k,2.7255/s.a,density/nb).map_err(ioerr)?;
        }
        writeln!(
            abundances,
            "{case},{},{},{},{},electron,0,,-1,{},,{}, {},BASELINE_H_HE_CHARGE_AND_BARYON_V1",
            p.time_s,
            1.0 / s.a - 1.0,
            eos.temperature_k,
            2.7255 / s.a,
            eos.electron_density_cm3,
            eos.electron_density_cm3 / nb,
            eos.electron_density_cm3 / nh
        )
        .map_err(ioerr)?;
        let gr = self.background.point_at_scale(1.0 / 21.0)?.geometry()?;
        let g = p.geometry()?;
        let mut angular = Vec::new();
        for cell in &self.source.angular_cells {
            let ray =
                CharacteristicRay::new(1.0, [(1.0 - cell.mu * cell.mu).sqrt(), 0.0, cell.mu], 0.0)?;
            angular.push(cell.count_fraction * ray.solid_angle_jacobian(&gr, &g)?);
        }
        let norm = angular.iter().sum::<f64>();
        for (k, weight) in angular.iter().enumerate() {
            let mu = self.ray(k, s.a)?.direction[2];
            for (i, &e) in self.energies.iter().enumerate() {
                let stock = s.photons[k * ne + i];
                // Fixed nodal representation: energy_low=energy_high=the node,
                // never mislabel these delta-node measures as cell densities.
                writeln!(
                    spectra,
                    "{case},{},{},{},{},{},{e},{e},{mu},{},{stock},{}",
                    self.config.q_ref,
                    p.sigma,
                    p.time_s,
                    s.a,
                    1.0 / s.a - 1.0,
                    4.0 * std::f64::consts::PI * weight / norm,
                    stock * e
                )
                .map_err(ioerr)?;
            }
        }
        let (nr, ur) = self.residuals(s)?;
        for (name, residual, tol) in [
            ("photon_count", nr, 1e-8 * s.ledgers.source_n.max(1.0)),
            (
                "augmented_energy_eV_per_H",
                ur,
                1e-8 * s.ledgers.source_u.max(1.0),
            ),
            ("J1_eV_per_H", m[1], 1e-12 * m[0].max(1e-30)),
        ] {
            writeln!(
                ledgers,
                "{case},{},{name},{residual},{tol},{}",
                p.time_s,
                if residual.abs() <= tol {
                    "PASS"
                } else {
                    "FAIL"
                }
            )
            .map_err(ioerr)?;
        }
        for (name, value) in [
            ("source_N_per_H", s.ledgers.source_n),
            ("source_U_eV_per_H", s.ledgers.source_u),
            ("absorbed_N_per_H", s.ledgers.absorbed_n),
            ("escape_U_eV_per_H", s.ledgers.escaped_u),
            ("boundary_N_per_H", s.ledgers.boundary_n),
            ("boundary_U_eV_per_H", s.ledgers.boundary_u),
            ("thermal_work_eV_per_H", s.ledgers.thermal_work),
            ("photon_work_eV_per_H", s.ledgers.photon_work),
            ("signed_CMB_input_eV_per_H", s.ledgers.cmb_input),
            ("DR_events_per_H", s.ledgers.dr_events),
            ("CE_cap_eV_per_H", s.ledgers.ce_cap_ev),
            ("excluded_DR_eV_per_H", s.ledgers.excluded_dr_ev),
        ] {
            writeln!(ledgers, "{case},{},{name},{value},,VALUE", p.time_s).map_err(ioerr)?;
        }
        for k in 0..3 {
            for (name, value) in [
                ("RR_events", s.ledgers.rr_events[k]),
                ("CI_events", s.ledgers.ci_events[k]),
                ("photo_events", s.ledgers.photo_events[k]),
                ("CI_floor_events", s.ledgers.ci_floor_events[k]),
            ] {
                writeln!(
                    ledgers,
                    "{case},{},{name}_{k}_per_H,{value},,VALUE",
                    p.time_s
                )
                .map_err(ioerr)?;
            }
        }
        let h0 = 67.4 * 1e5 / 3.0856775814913673e24;
        let g_si = 6.67430e-11;
        let c_si = 299792458.0;
        let coupling = 8.0 * std::f64::consts::PI * g_si / (3.0 * c_si * c_si);
        let eps_ref = h0 * h0 / coupling * 9.2e-5 / (1.0_f64 / 21.0).powi(4);
        let rad = isotropic_initial_radiation_moments(s.a, p.b, 1.0 / 21.0, 0.0, eps_ref, 4096)?;
        // erg/cm3 -> J/m3 is 0.1. eV/H -> erg/cm3 uses proper nH.
        let ion_stress = nh * m[2] * c.ev_erg * 0.1;
        let extra = nh * (m[0] * c.ev_erg + s.gas.w_erg_per_h) * 0.1
            + rad.signed_energy_correction_j_m3.abs();
        writeln!(
            feedback,
            "{case},{},{},{},{},{},{},{},{},{}",
            p.time_s,
            s.a,
            p.h_per_s,
            rad.pressure_anisotropy_j_m3,
            ion_stress,
            rad.signed_energy_correction_j_m3,
            extra,
            extra * coupling / p.h_flrw_per_s.powi(2),
            rad.pressure_anisotropy_j_m3 + ion_stress
        )
        .map_err(ioerr)?;
        Ok(())
    }
}
pub const SCIENCE_OUTPUT_BUDGET_BYTES: u64 = 224 * 1024 * 1024;
fn directory_bytes(out: &Path) -> Result<u64, ForwardError> {
    let mut total = 0_u64;
    for entry in fs::read_dir(out).map_err(ioerr)? {
        let entry = entry.map_err(ioerr)?;
        let metadata = entry.metadata().map_err(ioerr)?;
        let bytes = if metadata.is_dir() {
            directory_bytes(&entry.path())?
        } else {
            metadata.len()
        };
        total = total
            .checked_add(bytes)
            .ok_or(err("AXBI_OUTPUT_BYTES_OVERFLOW"))?;
    }
    Ok(total)
}
/// Scientific files have224MiB; launcher streams/envelope have32MiB reserved.
/// Reject before any batch write; include nested raw/checkpoint directories.
pub fn guard_output_bytes(out: &Path, new_bytes: u64) -> Result<(), ForwardError> {
    if directory_bytes(out)?
        .checked_add(new_bytes)
        .ok_or(err("AXBI_OUTPUT_BYTES_OVERFLOW"))?
        > SCIENCE_OUTPUT_BUDGET_BYTES
    {
        return Err(err("AXBI_OUTPUT_CAP"));
    }
    Ok(())
}
fn write_checkpoint(path: &Path, text: &str) -> Result<(), ForwardError> {
    let tmp = path.with_extension("tmp");
    fs::write(&tmp, text).map_err(ioerr)?;
    fs::rename(tmp, path).map_err(ioerr)
}
impl Driver {
    pub fn output_batch(
        &self,
        s: &DriverState,
        case: &str,
        out: &Path,
        save_spectrum: bool,
        h: &mut impl Write,
        a: &mut impl Write,
        p: &mut impl Write,
        l: &mut impl Write,
        fb: &mut impl Write,
    ) -> Result<(), ForwardError> {
        let (mut hv, mut av, mut pv, mut lv, mut fv) =
            (Vec::new(), Vec::new(), Vec::new(), Vec::new(), Vec::new());
        self.output_row(s, case, &mut hv, &mut av, &mut pv, &mut lv, &mut fv)?;
        let terminal_spectrum_reserve = if save_spectrum { 0 } else { pv.len() };
        if !save_spectrum {
            pv.clear();
        }
        h.flush().map_err(ioerr)?;
        a.flush().map_err(ioerr)?;
        p.flush().map_err(ioerr)?;
        l.flush().map_err(ioerr)?;
        fb.flush().map_err(ioerr)?;
        let bytes = hv.len() + av.len() + pv.len() + lv.len() + fv.len();
        // Reserve exact bytes for next snapshot plus atomic current checkpoint.
        let checkpoint_bytes = self.checkpoint_text(s)?.len();
        guard_output_bytes(
            out,
            (bytes + terminal_spectrum_reserve + 2 * checkpoint_bytes) as u64,
        )?;
        h.write_all(&hv).map_err(ioerr)?;
        a.write_all(&av).map_err(ioerr)?;
        p.write_all(&pv).map_err(ioerr)?;
        l.write_all(&lv).map_err(ioerr)?;
        fb.write_all(&fv).map_err(ioerr)?;
        Ok(())
    }
}

    pub fn spectrum_due(

    pub fn saved_spectrum_time(&self, out: &Path) -> Result<Option<f64>, ForwardError> {
        let text = fs::read_to_string(out.join("spectra.csv")).map_err(ioerr)?;
        let mut last = None;
        let mut rows = 0_usize;
        let full_rows = self.energies.len() * self.source.angular_cells.len();
        for line in text.lines().skip(1) {
            let t = line
                .split(',')
                .nth(3)
                .and_then(|x| x.parse::<f64>().ok())
                .ok_or(err("AXBI_SPECTRUM_TIME"))?;
            f(t)?;
            if Some(t) != last {
                if last.is_some() && rows != full_rows {
                    return Err(err("AXBI_SPECTRUM_SHAPE"));
                }
                if last.is_some_and(|previous| t <= previous) {
                    return Err(err("AXBI_SPECTRUM_ORDER"));
                }
                last = Some(t);
                rows = 0;
            }
            rows += 1;
        }
        if last.is_some() && rows != full_rows {
            return Err(err("AXBI_SPECTRUM_SHAPE"));
        }
        Ok(last)
    }
    /// A resume or controlled stop adds exactly one missing checkpoint-time
    /// full grid, never duplicates the main every-accepted-state tables.
    pub fn ensure_spectrum_snapshot(
        &self,
        s: &DriverState,
        case: &str,
        out: &Path,
        last: &mut Option<f64>,
        spectra: &mut impl Write,
    ) -> Result<bool, ForwardError> {
        let time = self.background.point_at_scale(s.a)?.time_s;
        if last.is_some_and(|t| t > time) {
            return Err(err("AXBI_SPECTRUM_AHEAD_CHECKPOINT"));
        }
        if *last == Some(time) {
            return Ok(false);
        }
        let mut pv = Vec::new();
        let (mut h, mut a, mut l, mut fb) = (
            std::io::sink(),
            std::io::sink(),
            std::io::sink(),
            std::io::sink(),
        );
        self.output_row(s, case, &mut h, &mut a, &mut pv, &mut l, &mut fb)?;
        spectra.flush().map_err(ioerr)?;
        let checkpoint_bytes = self.checkpoint_text(s)?.len();
        guard_output_bytes(out, (pv.len() + 2 * checkpoint_bytes) as u64)?;
        spectra.write_all(&pv).map_err(ioerr)?;
        *last = Some(time);
        Ok(true)
    }
}
fn open_output(
    path: &Path,
    resume: bool,
    header: &str,
) -> Result<BufWriter<fs::File>, ForwardError> {
    let exists = path.exists();
    let mut f = BufWriter::new(
        fs::OpenOptions::new()
            .create(true)
            .write(true)
            .append(resume)
            .truncate(!resume)
            .open(path)
            .map_err(ioerr)?,
    );
    if !resume || !exists {
        writeln!(f, "{header}").map_err(ioerr)?;
    }
    Ok(f)
}
/// Bound execution entry point. Actual science invocations are separately
/// admitted by the controller; unit tests do not run a cosmological history.
pub fn run(config_text: &str, out: &Path, case: &str, resume: bool) -> Result<(), ForwardError> {
    if case.is_empty() || case.contains(',') || case.contains('\n') || case.contains('\r') {
        return Err(err("AXBI_CASE_ID"));
    }
    let cfg = DriverConfig::read(config_text)?;
    let (driver, mut state) = Driver::new(cfg)?;
    if resume {
        if fs::read_to_string(out.join("config.cfg")).map_err(ioerr)? != config_text
            || fs::read_to_string(out.join("source_identity.txt")).map_err(ioerr)?
                != compiled_identity()
        {
            return Err(err("AXBI_RESUME_IDENTITY"));
        }
        if fs::read_to_string(out.join("case_id.txt")).map_err(ioerr)? != case {
            return Err(err("AXBI_RESUME_CASE"));
        }
        state = driver.read_checkpoint(&out.join("checkpoint.txt"))?;
        reconcile_outputs(out, driver.background.point_at_scale(state.a)?.time_s)?;
    } else {
        if out.exists() {
            return Err(err("AXBI_OUTPUT_EXISTS"));
        }
        fs::create_dir_all(out.join("raw")).map_err(ioerr)?;
        fs::create_dir(out.join("checkpoints")).map_err(ioerr)?;
        fs::write(out.join("config.cfg"), config_text).map_err(ioerr)?;
        fs::write(out.join("source_identity.txt"), compiled_identity()).map_err(ioerr)?;
        fs::write(out.join("case_id.txt"), case).map_err(ioerr)?;
        fs::write(out.join("metadata.txt"),"clock=elapsed proper seconds since zbar20\nframe=non-tilted normal tetrad\nmethod=binary64 BE split step-doubling empirical\nspectrum=cell-integrated source deposited to fixed physical-energy nodes; delta-node measures\nsolid_angle_weight_sr=normalized current quadrature for locally isotropic source; do not multiply stocks again\nspectra_cadence=initial; configured accepted-step interval(default60); each30wallsecondcheckpoint; terminal; fullgrid noaveraging\nuncertainty=no covariance/posterior supplied\ntail_tau=UNKNOWN\nscientific_admission=NOT_ISSUED\n").map_err(ioerr)?;
    }
    let mut h=open_output(&out.join("history.csv"),resume,"case_id,q_ref,Sigma,time_s,a,zbar,b,H_per_s,s_per_s,x_HII,x_HeII,x_HeIII,nH_proper_cm3,ne_proper_cm3,temperature_K,Gamma_HI_per_s,Gamma_HeI_per_s,Gamma_HeII_per_s,photon_n_per_H,photon_u_eV_per_H,J0_photon_u_eV_per_H,J1_photon_u_eV_per_H,J2_photon_u_eV_per_H,x_HI,x_HeI")?;
    let mut a=open_output(&out.join("abundances.csv"),resume,"run_id,time_s,zbar,Tgas_K,background_Tref_K,species_id,A,Z,charge,nproper,element_fraction,Ybaryon,electron_xe,normalization_registry")?;
    let mut p=open_output(&out.join("spectra.csv"),resume,"case_id,q_ref,Sigma,time_s,a,zbar,energy_low_eV,energy_high_eV,mu_center,solid_angle_weight_sr,photon_n_per_H,photon_u_eV_per_H")?;
    let mut l = open_output(
        &out.join("ledgers.csv"),
        resume,
        "case_id,time_s,ledger_name,residual,tolerance,status",
    )?;
    let mut fb=open_output(&out.join("feedback.csv"),resume,"case_id,time_s,a,H_per_s,effective_radiation_DeltaP_J_m3,ionizing_DeltaP_J_m3,effective_radiation_signed_deltaepsilon_J_m3,extra_energy_J_m3,extra_energy_ratio,total_DeltaP_J_m3")?;
    let mut rejects = open_output(
        &out.join("raw/rejections.csv"),
        true,
        "attempt,a,dln_a,error",
    )?;
    let mut last_spectrum_time = if resume {
        driver.saved_spectrum_time(out)?
    } else {
        None
    };
    if !resume {
        driver.output_batch(
            &state, case, out, true, &mut h, &mut a, &mut p, &mut l, &mut fb,
        )?;
        last_spectrum_time = Some(driver.background.point_at_scale(state.a)?.time_s);
    }
    if !resume {
        for writer in [&mut h, &mut a, &mut p, &mut l, &mut fb, &mut rejects] {
            writer.flush().map_err(ioerr)?;
        }
        driver.save_checkpoint_budget(out, &out.join("checkpoint.txt"), &state)?;
    }
    if resume {
        driver.ensure_spectrum_snapshot(&state, case, out, &mut last_spectrum_time, &mut p)?;
    }
    let started = Instant::now();
    let prior_wall = state.wall_elapsed_s;
    let mut last_save = Instant::now();
    let end = 1.0 / (1.0 + driver.config.end_zbar);
    let outcome = (|| {
        while state.a < end {
            state.wall_elapsed_s = prior_wall + started.elapsed().as_secs_f64();
            if state.wall_elapsed_s >= driver.config.max_wall_s
                || state.accepted >= driver.config.max_steps
                || state.attempts >= driver.config.max_attempts
                || state.rhs_evaluations >= driver.config.max_rhs
            {
                return Err(err("AXBI_RUNTIME_CAP"));
            }
            let previous = state.clone();
            let accepted = match driver.adaptive_step(&mut state) {
                Ok(value) => value,
                Err(e) => {
                    writeln!(
                        rejects,
                        "{},{},{},{e:?}",
                        state.attempts, state.a, state.next_dln_a
                    )
                    .map_err(ioerr)?;
                    rejects.flush().map_err(ioerr)?;
                    if state.consecutive_rejects >= 30
                        || state.next_dln_a < 1e-10
                        || state.rhs_evaluations >= driver.config.max_rhs
                    {
                        return Err(e);
                    }
                    false
                }
            };
            if !accepted {
                writeln!(
                    rejects,
                    "{},{},{},local_metric={}",
                    state.attempts, state.a, state.next_dln_a, state.last_error
                )
                .map_err(ioerr)?;
                if state.consecutive_rejects >= 30 || state.next_dln_a < 1e-10 {
                    return Err(err("AXBI_REJECT_CAP"));
                }
                continue;
            }
            let checkpoint_due = driver.spectrum_due(
                state.accepted,
                last_save.elapsed().as_secs_f64(),
                state.a == end,
            );
            if let Err(error) = driver.output_batch(
                &state,
                case,
                out,
                checkpoint_due,
                &mut h,
                &mut a,
                &mut p,
                &mut l,
                &mut fb,
            ) {
                let failed = state.clone();
                state = previous;
                state.attempts = failed.attempts;
                state.comparisons = failed.comparisons;
                state.rhs_evaluations = failed.rhs_evaluations;
                state.rejects = failed.rejects;
                state.wall_elapsed_s = failed.wall_elapsed_s;
                return Err(error);
            }
            if checkpoint_due {
                last_spectrum_time = Some(driver.background.point_at_scale(state.a)?.time_s);
                for writer in [&mut h, &mut a, &mut p, &mut l, &mut fb, &mut rejects] {
                    writer.flush().map_err(ioerr)?;
                }
                state.wall_elapsed_s = prior_wall + started.elapsed().as_secs_f64();
                driver.save_checkpoint_budget(
                    out,
                    &out.join(format!("checkpoints/{:08}.txt", state.accepted)),
                    &state,
                )?;
                driver.save_checkpoint_budget(out, &out.join("checkpoint.txt"), &state)?;
                check_resources(out)?;
                last_save = Instant::now();
            }
        }
        Ok(())
    })();
    state.wall_elapsed_s = prior_wall + started.elapsed().as_secs_f64();
    for writer in [&mut h, &mut a, &mut p, &mut l, &mut fb, &mut rejects] {
        writer.flush().map_err(ioerr)?;
    }
    driver.ensure_spectrum_snapshot(&state, case, out, &mut last_spectrum_time, &mut p)?;
    p.flush().map_err(ioerr)?;
    driver.save_checkpoint_budget(out, &out.join("checkpoint.txt"), &state)?;
    summarize_feedback(out)?;
    fs::write(out.join("RETURN.txt"),format!("status={}\na={}\naccepted={}\nattempts={}\nrejections={}\nrhs_evaluations={}\nwall_s={}\nmax_newton_residual={}\nlast_error={}\nresult={outcome:?}\nscientific_admission=NOT_ISSUED\n",if outcome.is_ok(){"COMPLETE_REPRESENTED_INTERVAL"}else{"STOPPED_CHECKPOINTED"},state.a,state.accepted,state.attempts,state.rejects,state.rhs_evaluations,state.wall_elapsed_s,state.max_newton_residual,state.last_error)).map_err(ioerr)?;
    outcome
}
fn check_resources(out: &Path) -> Result<(), ForwardError> {
    let mut bytes = 0_u64;
    for e in fs::read_dir(out).map_err(ioerr)? {
        let e = e.map_err(ioerr)?;
        if e.path().is_file() {
            bytes += e.metadata().map_err(ioerr)?.len();
        }
    }
    if bytes > 256 * 1024 * 1024 {
        return Err(err("AXBI_OUTPUT_CAP"));
    }
    if let Ok(status) = fs::read_to_string("/proc/self/status") {
        if let Some(line) = status.lines().find(|s| s.starts_with("VmHWM:")) {
            if line
                .split_whitespace()
                .nth(1)
                .and_then(|s| s.parse::<u64>().ok())
                .unwrap_or(0)
                > 2 * 1024 * 1024
            {
                return Err(err("AXBI_RSS_CAP"));
            }
        }
    }
    Ok(())
}
/// Restore tables to the durable checkpoint; preserve discarded rows as raw
/// crash-tail evidence. Never append later rows to an earlier resumed state.
fn reconcile_outputs(out: &Path, time: f64) -> Result<(), ForwardError> {
    for (name, time_col) in [
        ("history.csv", 3),
        ("spectra.csv", 3),
        ("abundances.csv", 1),
        ("ledgers.csv", 1),
        ("feedback.csv", 1),
    ] {
        let file = out.join(name);
        let text = fs::read_to_string(&file).map_err(ioerr)?;
        let mut kept = String::new();
        let mut tail = String::new();
        for (i, line) in text.lines().enumerate() {
            let t = if i == 0 {
                0.0
            } else {
                line.split(',')
                    .nth(time_col)
                    .and_then(|s| s.parse::<f64>().ok())
                    .ok_or(err("AXBI_RESUME_TABLE"))?
            };
            if i == 0 || t <= time {
                kept.push_str(line);
                kept.push('\n');
            } else {
                tail.push_str(line);
                tail.push('\n');
            }
        }
        if !tail.is_empty() {
            let epoch = std::time::SystemTime::now()
                .duration_since(std::time::UNIX_EPOCH)
                .map_err(|_| err("AXBI_SYSTEM_CLOCK"))?
                .as_nanos();
            fs::write(out.join(format!("raw/crash_tail_{epoch}_{name}")), tail).map_err(ioerr)?;
            fs::write(file, kept).map_err(ioerr)?;
        }
    }
    Ok(())
}
fn summarize_feedback(out: &Path) -> Result<(), ForwardError> {
    let text = fs::read_to_string(out.join("feedback.csv")).map_err(ioerr)?;
    let coupling = 8.0 * std::f64::consts::PI * 6.67430e-11 / (3.0 * 299792458.0_f64.powi(2));
    let mut prev = None;
    let mut integral = 0.0;
    let mut max_ds = 0.0_f64;
    let mut max_energy = 0.0_f64;
    for line in text.lines().skip(1) {
        let cols = line.split(',').collect::<Vec<_>>();
        if cols.len() != 10 {
            return Err(err("AXBI_FEEDBACK_TABLE"));
        }
        let get = |i: usize| {
            cols[i]
                .parse::<f64>()
                .map_err(|_| err("AXBI_FEEDBACK_TABLE"))
        };
        let t = get(1)?;
        let a = get(2)?;
        let h = get(3)?;
        let stress = get(9)?;
        max_energy = max_energy.max(get(8)?);
        if let Some((pt, pa, ps)) = prev {
            integral += 0.5 * coupling * (a.powi(3) * stress + pa * ps) * (t - pt);
        }
        max_ds = max_ds.max((integral / a.powi(3) / h).abs());
        prev = Some((t, a.powi(3), stress));
    }
    fs::write(out.join("feedback_summary.txt"),format!("max_abs_delta_s_over_H={max_ds}\nmax_extra_energy_ratio={max_energy}\nthreshold=0.0001\nstatus={}\nkind=sampled_trapezoidal_perturbative_estimate_not_rigorous_bound\n",if max_ds<=1e-4&&max_energy<=1e-4{"WITHIN_DECLARED_DIAGNOSTIC_CEILING"}else{"EXCEEDS_DECLARED_DIAGNOSTIC_CEILING"})).map_err(ioerr)
}

AXBI_SOURCE_BOUNDARY
//! Stress-free, density-fixed axisymmetric Bianchi I in elapsed proper seconds.
//! Parameters have no physical defaults; SAME_H is an explicit counterfactual.
use crate::{AxisymmetricPoint, ForwardError, GeometryBackground, GeometrySnapshot};

pub const DEFAULT_PANELS: usize = 4096;
fn invalid(tag: &'static str) -> ForwardError {
    ForwardError::InvalidInput(tag)
}
fn finite(x: f64) -> Result<f64, ForwardError> {
    if x.is_finite() {
        Ok(x)
    } else {
        Err(invalid("AxbiNonFinite"))
    }
}
fn positive(x: f64) -> Result<f64, ForwardError> {
    if finite(x)? > 0.0 {
        Ok(x)
    } else {
        Err(invalid("AxbiPositive"))
    }
}
#[derive(Clone, Copy, Debug, PartialEq)]
pub struct Cosmology {
    pub h0_per_s: f64,
    pub omega_m: f64,
    pub omega_r: f64,
    pub omega_lambda: f64,
}
impl Cosmology {
    pub fn hubble_flrw(&self, a: f64) -> Result<f64, ForwardError> {
        positive(a)?;
        positive(self.h0_per_s)?;
        for density in [self.omega_m, self.omega_r, self.omega_lambda] {
            if finite(density)? < 0.0 {
                return Err(invalid("AxbiDensity"));
            }
        }
        positive(
            self.h0_per_s
                * positive(
                    self.omega_m / a.powi(3) + self.omega_r / a.powi(4) + self.omega_lambda,
                )?
                .sqrt(),
        )
    }
}
#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum BackgroundFamily {
    PhysicalDensityFixed,
    SameHDiagnostic,
}
#[derive(Clone, Copy, Debug, PartialEq)]
pub struct BackgroundPoint {
    pub a: f64,
    pub b: f64,
    pub time_s: f64,
    pub h_flrw_per_s: f64,
    pub h_per_s: f64,
    pub s_per_s: f64,
    /// Instantaneous s/H, distinct from the fixed signed q_ref.
    pub sigma: f64,
}
impl BackgroundPoint {
    pub fn geometry(&self) -> Result<GeometrySnapshot, ForwardError> {
        AxisymmetricPoint::new(self.a, self.b, self.h_per_s, self.s_per_s)?.snapshot()
    }
    pub fn sigma_squared_per_s2(&self) -> f64 {
        3.0 * self.s_per_s * self.s_per_s
    }
}
#[derive(Clone, Debug)]
pub struct AxbiBackground {
    cosmology: Cosmology,
    a_ref: f64,
    a_end: f64,
    b_ref: f64,
    q_ref: f64,
    family: BackgroundFamily,
    panels: usize,
}
impl AxbiBackground {
    pub fn new(
        cosmology: Cosmology,
        a_ref: f64,
        a_end: f64,
        b_ref: f64,
        q_ref: f64,
        family: BackgroundFamily,
        panels: usize,
    ) -> Result<Self, ForwardError> {
        positive(a_ref)?;
        positive(a_end)?;
        finite(b_ref)?;
        finite(q_ref)?;
        if a_end <= a_ref || !(2..=65536).contains(&panels) || panels % 2 != 0 {
            return Err(invalid("AxbiDomainOrPanels"));
        }
        cosmology.hubble_flrw(a_ref)?;
        cosmology.hubble_flrw(a_end)?;
        let result = Self {
            cosmology,
            a_ref,
            a_end,
            b_ref,
            q_ref,
            family,
            panels,
        };
        result.point_at_scale(a_end)?;
        Ok(result)
    }
    pub fn q_ref(&self) -> f64 {
        self.q_ref
    }
    pub fn family(&self) -> BackgroundFamily {
        self.family
    }
    fn rates(&self, a: f64) -> Result<(f64, f64, f64), ForwardError> {
        let hf = self.cosmology.hubble_flrw(a)?;
        let s = finite(
            self.q_ref * self.cosmology.hubble_flrw(self.a_ref)? * (self.a_ref / a).powi(3),
        )?;
        let h = match self.family {
            BackgroundFamily::PhysicalDensityFixed => positive(hf.hypot(s))?,
            BackgroundFamily::SameHDiagnostic => hf,
        };
        Ok((hf, h, s))
    }
    pub fn point_at_scale(&self, a: f64) -> Result<BackgroundPoint, ForwardError> {
        positive(a)?;
        if a < self.a_ref || a > self.a_end {
            return Err(invalid("AxbiScaleDomain"));
        }
        let span = (a / self.a_ref).ln();
        let dx = span / self.panels as f64;
        let (mut clock, mut anisotropy) = (0.0, 0.0);
        for i in 0..=self.panels {
            let ai = self.a_ref * (dx * i as f64).exp();
            let (_, h, s) = self.rates(ai)?;
            let w = if i == 0 || i == self.panels {
                1.0
            } else if i % 2 == 0 {
                2.0
            } else {
                4.0
            };
            clock += w / h;
            anisotropy += w * s / h;
        }
        let (hf, h, s) = self.rates(a)?;
        let point = BackgroundPoint {
            a,
            b: finite(self.b_ref + anisotropy * dx / 3.0)?,
            time_s: finite(clock * dx / 3.0)?,
            h_flrw_per_s: hf,
            h_per_s: h,
            s_per_s: s,
            sigma: finite(s / h)?,
        };
        point.geometry()?;
        Ok(point)
    }
    pub fn point_at_time(&self, time_s: f64) -> Result<BackgroundPoint, ForwardError> {
        finite(time_s)?;
        let end = self.point_at_scale(self.a_end)?;
        if time_s < 0.0 || time_s > end.time_s {
            return Err(invalid("AxbiTimeDomain"));
        }
        if time_s == 0.0 {
            return self.point_at_scale(self.a_ref);
        }
        if time_s == end.time_s {
            return Ok(end);
        }
        let (mut lo, mut hi) = (self.a_ref.ln(), self.a_end.ln());
        for _ in 0..64 {
            let mid = (lo + hi) * 0.5;
            if self.point_at_scale(mid.exp())?.time_s < time_s {
                lo = mid;
            } else {
                hi = mid;
            }
        }
        self.point_at_scale(((lo + hi) * 0.5).exp())
    }
}
impl GeometryBackground for AxbiBackground {
    fn snapshot(&self, t: f64) -> Result<GeometrySnapshot, ForwardError> {
        self.point_at_time(t)?.geometry()
    }
}

/// Actual omitted-stress history: SI pressure/energy, elapsed proper seconds.
#[derive(Clone, Copy, Debug)]
pub struct FeedbackSample {
    pub a: f64,
    pub time_s: f64,
    pub pressure_anisotropy_j_m3: f64,
    pub extra_energy_j_m3: f64,
}
#[derive(Clone, Copy, Debug)]
pub struct FeedbackDiagnostic {
    pub max_abs_delta_s_over_h: f64,
    pub max_extra_energy_over_background: f64,
    /// This is a sampled trapezoidal perturbative estimate, not a rigorous bound.
    pub endpoint_delta_s_per_s: f64,
}
/// No stress history is inferred from q. Include free-streaming CMB stress as
/// well as ionizing photons. G and c must be supplied in SI units.
pub fn omitted_feedback(
    background: &AxbiBackground,
    samples: &[FeedbackSample],
    g_m3_kg_s2: f64,
    c_m_s: f64,
) -> Result<FeedbackDiagnostic, ForwardError> {
    positive(g_m3_kg_s2)?;
    positive(c_m_s)?;
    if samples.len() < 2 {
        return Err(invalid("AxbiFeedbackCoverage"));
    }
    if samples[0].a != background.a_ref || samples[0].time_s != 0.0 {
        return Err(invalid("AxbiFeedbackOrigin"));
    }
    let k = positive(8.0 * std::f64::consts::PI * g_m3_kg_s2 / (3.0 * c_m_s * c_m_s))?;
    let mut integral = 0.0;
    let mut diagnostic = FeedbackDiagnostic {
        max_abs_delta_s_over_h: 0.0,
        max_extra_energy_over_background: 0.0,
        endpoint_delta_s_per_s: 0.0,
    };
    for (i, sample) in samples.iter().enumerate() {
        let point = background.point_at_scale(sample.a)?;
        finite(sample.time_s)?;
        finite(sample.pressure_anisotropy_j_m3)?;
        if finite(sample.extra_energy_j_m3)? < 0.0 {
            return Err(invalid("AxbiFeedbackEnergy"));
        }
        if (sample.time_s - point.time_s).abs() > 1e-7 * point.time_s.max(1.0) {
            return Err(invalid("AxbiFeedbackClock"));
        }
        if i > 0 {
            let prev = &samples[i - 1];
            if sample.a <= prev.a || sample.time_s <= prev.time_s {
                return Err(invalid("AxbiFeedbackOrder"));
            }
            integral = finite(
                integral
                    + 0.5
                        * k
                        * (sample.a.powi(3) * sample.pressure_anisotropy_j_m3
                            + prev.a.powi(3) * prev.pressure_anisotropy_j_m3)
                        * (sample.time_s - prev.time_s),
            )?;
        }
        let ds = finite(integral / sample.a.powi(3))?;
        diagnostic.max_abs_delta_s_over_h = diagnostic
            .max_abs_delta_s_over_h
            .max(finite(ds.abs() / point.h_per_s)?);
        // Total isotropic background including Lambda: H_FLRW²/k.
        diagnostic.max_extra_energy_over_background =
            diagnostic.max_extra_energy_over_background.max(finite(
                sample.extra_energy_j_m3 * k / point.h_flrw_per_s.powi(2),
            )?);
        diagnostic.endpoint_delta_s_per_s = ds;
    }
    Ok(diagnostic)
}

/// Free-streamed initially isotropic massless-radiation moments, SI units.
#[derive(Clone, Copy, Debug)]
pub struct IsotropicRadiationMoments {
    pub energy_j_m3: f64,
    pub pressure_anisotropy_j_m3: f64,
    pub signed_energy_correction_j_m3: f64,
    pub dipole_energy_j_m3: f64,
}
/// Integrate initial directions mu0, not the evolved local angular measure.
/// The full effective Omega_r fluid may be used as a conservative free-stream
/// envelope; this is a perturbative diagnostic rather than coupled evolution.
pub fn isotropic_initial_radiation_moments(
    a: f64,
    b: f64,
    a_ref: f64,
    b_ref: f64,
    epsilon_ref_j_m3: f64,
    panels: usize,
) -> Result<IsotropicRadiationMoments, ForwardError> {
    positive(a)?;
    positive(a_ref)?;
    finite(b)?;
    finite(b_ref)?;
    if finite(epsilon_ref_j_m3)? < 0.0 || !(2..=65536).contains(&panels) || panels % 2 != 0 {
        return Err(invalid("AxbiRadiationDomain"));
    }
    let db = finite(b - b_ref)?;
    let ep = positive((2.0 * db).exp())?;
    let ez = positive((-4.0 * db).exp())?;
    let mut monopole = 0.0;
    let mut quadrupole = 0.0;
    for i in 0..=panels {
        let mu = i as f64 / panels as f64;
        let r = positive(((1.0 - mu * mu) * ep + mu * mu * ez).sqrt())?;
        let local_mu = mu * ez.sqrt() / r;
        let w = if i == 0 || i == panels {
            1.0
        } else if i % 2 == 0 {
            2.0
        } else {
            4.0
        };
        monopole += w * r;
        quadrupole += w * r * (3.0 * local_mu * local_mu - 1.0) / 2.0;
    }
    let eps_iso = finite(epsilon_ref_j_m3 * (a_ref / a).powi(4))?;
    let norm = 3.0 * panels as f64;
    let energy = finite(eps_iso * monopole / norm)?;
    Ok(IsotropicRadiationMoments {
        energy_j_m3: energy,
        pressure_anisotropy_j_m3: finite(eps_iso * quadrupole / norm)?,
        signed_energy_correction_j_m3: finite(energy - eps_iso)?,
        dipole_energy_j_m3: 0.0,
    })
}

fn invalid() -> ForwardError {
    ForwardError::InvalidInput("AXBI_SOURCE_DOMAIN_OR_QUADRATURE")
}
fn positive(x: f64) -> Result<f64, ForwardError> {
    if x.is_normal() && x > 0.0 {
        Ok(x)
    } else {
        Err(invalid())
    }
}
fn shape(energy_ev: f64) -> f64 {
    let c = HHeModel::controlled_fixture();
    let kt = c.kb_erg_k / c.ev_erg * 50000.0;
    energy_ev * energy_ev / (energy_ev / kt).exp_m1()
}
fn simpson(lo: f64, hi: f64, n: usize, energy_moment: bool) -> f64 {
    let step = (hi - lo) / n as f64;
    let mut sum = 0.0;
    for i in 0..=n {
        let e = lo + step * i as f64;
        let f = shape(e) * if energy_moment { e } else { 1.0 };
        sum += f * if i == 0 || i == n {
            1.0
        } else if i % 2 == 0 {
            2.0
        } else {
            4.0
        };
    }
    sum * step / 3.0
}
fn angular(n: usize) -> Result<Vec<AngularCell>, ForwardError> {
    if ![8, 16, 32].contains(&n) {
        return Err(invalid());
    }
    let mut out = vec![
        AngularCell {
            mu: 0.0,
            count_fraction: 0.0
        };
        n
    ];
    for i in 0..n / 2 {
        let mut x = (std::f64::consts::PI * (i as f64 + 0.75) / (n as f64 + 0.5)).cos();
        let mut converged = false;
        for _ in 0..32 {
            let (p, pm) = legendre(n, x);
            let d = n as f64 * (x * p - pm) / (x * x - 1.0);
            let next = x - p / d;
            if (next - x).abs() <= 2.0 * f64::EPSILON {
                x = next;
                converged = true;
                break;
            }
            x = next;
        }
        if !converged {
            return Err(invalid());
        }
        let (p, pm) = legendre(n, x);
        let d = n as f64 * (x * p - pm) / (x * x - 1.0);
        let weight = positive(1.0 / ((1.0 - x * x) * d * d))?;
        out[i] = AngularCell {
            mu: -x,
            count_fraction: weight,
        };
        out[n - 1 - i] = AngularCell {
            mu: x,
            count_fraction: weight,
        };
    }
    Ok(out)
}
fn legendre(n: usize, x: f64) -> (f64, f64) {
    let (mut pm, mut p) = (1.0, x);
    for k in 2..=n {
        let q = ((2 * k - 1) as f64 * x * p - (k - 1) as f64 * pm) / k as f64;
        pm = p;
        p = q;
    }
    (p, pm)
}
impl AxbiSource {
    /// Allowed source sensitivity fesc is explicit; baseline .2, variant .25.
    /// nH0 is the a=1 comoving hydrogen density in cm^-3, not an endpoint proper
    /// density. Baseline cells_per_segment=16, n_mu=8; resolutions are numeric.
    pub fn new(
        n_h0_cm3: f64,
        escape_fraction: f64,
        cells_per_segment: usize,
        n_mu: usize,
    ) -> Result<Self, ForwardError> {
        positive(n_h0_cm3)?;
        positive(escape_fraction)?;
        if escape_fraction > 1.0 || !(1..=1024).contains(&cells_per_segment) {
            return Err(invalid());
        }
        let edges = [13.6_f64, 24.59, 54.42, 100.0];
        let mut integrals = Vec::new();
        for segment in edges.windows(2) {
            for j in 0..cells_per_segment {
                let lo = if j == 0 {
                    segment[0]
                } else {
                    segment[0] * (segment[1] / segment[0]).powf(j as f64 / cells_per_segment as f64)
                };
                let hi = if j + 1 == cells_per_segment {
                    segment[1]
                } else {
                    segment[0]
                        * (segment[1] / segment[0]).powf((j + 1) as f64 / cells_per_segment as f64)
                };
                let count = positive(simpson(lo, hi, 32, false))?;
                let energy = positive(simpson(lo, hi, 32, true))?;
                integrals.push((lo, hi, count, energy));
            }
        }
        let normalization = positive(integrals.iter().map(|x| x.2).sum())?;
        let reference = positive(simpson(edges[0], edges[3], 4096, false))?;
        let normalization_relative_residual = (normalization / reference - 1.0).abs();
        if normalization_relative_residual > 1e-7 {
            return Err(invalid());
        }
        let spectral_cells = integrals
            .into_iter()
            .map(|(lower_ev, upper_ev, count, energy)| SpectralCell {
                lower_ev,
                upper_ev,
                energy_ev: energy / count,
                photon_fraction: count / normalization,
            })
            .collect();
        let tail = positive(simpson(100.0, 500.0, 4096, false))?;
        Ok(Self {
            spectral_cells,
            angular_cells: angular(n_mu)?,
            normalization_relative_residual,
            omitted_blackbody_tail_photon_fraction: tail / (reference + tail),
            n_h0_cm3,
            escape_fraction,
        })
    }
    /// Continuous total emissivity. No year conversion is applied: MD14 psi is
    /// Msun/yr/cMpc^3 and the yield is photons/s per (Msun/yr).
    pub fn at_redshift(&self, zbar: f64) -> Result<EmissionRate, ForwardError> {
        if !zbar.is_finite() || !(4.0..=20.0).contains(&zbar) {
            return Err(invalid());
        }
        let zp = 1.0 + zbar;
        let sfr = 0.015 * zp.powf(2.7) / (1.0 + (zp / 2.9).powf(5.6));
        let log_yield = 1.0 / (0.00038 * 0.002_f64.powf(0.227) + 0.01858) - 2.55_f64.log10();
        let comoving = positive(self.escape_fraction * 10_f64.powf(log_yield) * sfr)?;
        let per_volume = positive(comoving / MPC_CM.powi(3))?;
        Ok(EmissionRate {
            comoving_photons_per_s_cmpc3: comoving,
            proper_photons_cm3_s: positive(per_volume * zp.powi(3))?,
            per_h_s: positive(per_volume / self.n_h0_cm3)?,
        })
    }
    /// Source-only derivative for one discrete photon population, H^-1 s^-1.
    /// Driver time integration and photon evolution remain separate owners.
    pub fn cell_per_h_s(
        &self,
        zbar: f64,
        energy_index: usize,
        angle_index: usize,
    ) -> Result<f64, ForwardError> {
        let e = self.spectral_cells.get(energy_index).ok_or_else(invalid)?;
        let a = self.angular_cells.get(angle_index).ok_or_else(invalid)?;
        positive(self.at_redshift(zbar)?.per_h_s * e.photon_fraction * a.count_fraction)
    }
}

/// Operational package domain, not empirical support. No clamping of T.

pub fn igm_rates(t: f64) -> Result<IgmRates, ForwardError> {
    if !t.is_finite() || !(1.0..=1e6).contains(&t) {
        return Err(ForwardError::InvalidInput("IGM_TEMPERATURE_DOMAIN"));
    }
    use Process::*;
    let c = |p| source_coefficient(p, t);
    let ci = [c(K1)?, c(K3)?, c(K5)?];
    let te = t / TEVK;
    let low = te <= 0.8;
    let (rr_he, dr) = he_recombination(t);
    let ce = [c(CeHI)?, c(CeHeI)?, c(CeHeII)?];
    let dr_raw = c(ReHeII2)?;
    Ok(IgmRates {
        ci_cm3_s: ci,
        rr_cm3_s: [c(K2)?, rr_he, c(K6)?],
        dr_cm3_s: dr,
        ce_erg: ce,
        rr_cooling_erg_cm3_s: [c(ReHII)?, c(ReHeII1)?, c(ReHeIII)?],
        dr_cooling_erg_cm3_s: if low { 0.0 } else { dr_raw },
        dr_raw_cooling_erg_cm3_s: dr_raw,
        freefree_erg_cm3_s: c(Brem)?,
        diagnostics: IgmRateDiagnostics {
            ci_floor_cm3_s: std::array::from_fn(|i| if low && ci[i] == TINY { ci[i] } else { 0.0 }),

                    ce[i]
                } else {

                }
            }),

// Preserve the original C source literals, including excess decimal digits.
#[allow(clippy::excessive_precision)]
fn source_coefficient(process: Process, t: f64) -> Result<f64, ForwardError> {
    if !t.is_finite() || !(1.0..=1e6).contains(&t) {
        return Err(ForwardError::InvalidInput("IGM_TEMPERATURE_DOMAIN"));
    }
    use Process::*;
    let te = t / TEVK;
    let l = te.ln();
    let k1 = || {
        let v = (-32.71396786375 + 13.53655609057 * l - 5.739328757388 * l.powi(2)
            + 1.563154982022 * l.powi(3)
            - 0.2877056004391 * l.powi(4)
            + 0.03482559773736999 * l.powi(5)
            - 0.00263197617559 * l.powi(6)
            + 0.0001119543953861 * l.powi(7)
            - 2.039149852002e-6 * l.powi(8))
        .exp();
        if te <= 0.8 {
            v.max(TINY)
        } else {
            v
        }
    };
    let k3 = || {
        if te > 0.8 {
            (-44.09864886561001 + 23.91596563469 * l - 10.75323019821 * l.powi(2)
                + 3.058038757198 * l.powi(3)
                - 0.5685118909884001 * l.powi(4)
                + 0.06795391233790001 * l.powi(5)
                - 0.005009056101857001 * l.powi(6)
                + 0.0002067236157507 * l.powi(7)
                - 3.649161410833e-6 * l.powi(8))
            .exp()
        } else {
            TINY
        }
    };
    let k5 = || {
        if te > 0.8 {
            (-68.71040990212001 + 43.93347632635 * l - 18.48066993568 * l.powi(2)
                + 4.701626486759002 * l.powi(3)
                - 0.7692466334492 * l.powi(4)
                + 0.08113042097303 * l.powi(5)
                - 0.005324020628287001 * l.powi(6)
                + 0.0001975705312221 * l.powi(7)
                - 3.165581065665e-6 * l.powi(8))
            .exp()
        } else {
            TINY
        }
    };
    let v = match process {
        K1 => k1(),
        K2 => {
            if t > 5500.0 {
                (-28.61303380689232
                    - 0.7241125657826851 * l
                    - 0.02026044731984691 * l.powi(2)
                    - 0.002380861877349834 * l.powi(3)
                    - 0.0003212605213188796 * l.powi(4)
                    - 0.00001421502914054107 * l.powi(5)
                    + 4.989108920299513e-6 * l.powi(6)
                    + 5.755614137575758e-7 * l.powi(7)
                    - 1.856767039775261e-8 * l.powi(8)
                    - 3.071135243196595e-9 * l.powi(9))
                .exp()
            } else {
                he_recombination(t).0
            }
        }
        K3 => k3(),
        K5 => k5(),
        K6 => 3.36e-10 / t.sqrt() / (t / 1e3).powf(0.2) / (1.0 + (t / 1e6).powf(0.7)),
        CeHI => 7.5e-19 * (-(118348.0 / t).min(LN_DHUGE)).exp() / (1.0 + (t / 1e5).sqrt()),
        CeHeI => {
            9.1e-27 * (-(13179.0 / t).min(LN_DHUGE)).exp() * t.powf(-0.1687)
                / (1.0 + (t / 1e5).sqrt())
        }
        CeHeII => {
            5.54e-17 * (-(473638.0 / t).min(LN_DHUGE)).exp() * t.powf(-0.3970)
                / (1.0 + (t / 1e5).sqrt())
        }
        ReHII => {
            let lambda = 2.0 * 157807.0 / t;

            1.778e-29 * t * lambda.powf(1.965) / (1.0 + (lambda / 0.541).powf(0.502)).powf(2.697)
        }
        ReHeII1 => {
            let lambda = 2.0 * 285335.0 / t;

            3e-14 * KBOLTZ * t * lambda.powf(0.654)
        }
        ReHeII2 => {
            1.24e-13
                * t.powf(-1.5)
                * (-(470000.0 / t).min(LN_DHUGE)).exp()
                * (1.0 + 0.3 * (-(94000.0 / t).min(LN_DHUGE)).exp())
        }
        ReHeIII => {
            let lambda = 2.0 * 631515.0 / t;
            // Preserve original raw 8*T term; no kinetic-moment substitution.

            8.0 * 1.778e-29 * t * lambda.powf(1.965)
                / (1.0 + (lambda / 0.541).powf(0.502)).powf(2.697)
        }
        Brem => 1.43e-27 * t.sqrt() * (1.1 + 0.34 * (-(5.5 - t.log10()).powi(2) / 3.0).exp()),
    };
    if !v.is_finite() || v < 0.0 {
        return Err(ForwardError::InvalidInput("IGM_COEFFICIENT_INVALID"));
    }
    Ok(v)
}

// The two k4 summands, evaluated separately to avoid cancellation in tiny DR.
fn he_recombination(t: f64) -> (f64, f64) {
    let te = t / TEVK;
    let rr = 3.92e-13 / te.powf(0.6353);
    let dr = if te <= 0.8 {
        0.0
    } else {
        1.54e-9 * (1.0 + 0.3 / (8.099328789667 / te).exp())
            / ((40.49664394833662 / te).exp() * te.powf(1.5))
    };
    (rr, dr)
}

use crate::{igm_rates::igm_rates, HHeModel};
// CODATA 2022: https://physics.nist.gov/cuu/pdf/wall_2022.pdf
// sigma_T and electron mass; a_rad = 4*sigma_SB/c.
// sigma_SB follows the exact SI kB,h,c blackbody expression (converted to cgs).
const SIGMA_T_CM2: f64 = 6.6524587051e-25;
const ELECTRON_MASS_G: f64 = 9.1093837139e-28;
const PLANCK_ERG_S: f64 = 6.62607015e-27;
fn invalid() -> ForwardError {
    ForwardError::InvalidInput("IGM_RHS_DOMAIN_OR_ARITHMETIC")
}
fn finite(x: f64) -> Result<f64, ForwardError> {
    if x == 0.0 || x.is_normal() {
        Ok(x)
    } else {
        Err(invalid())
    }
}
fn nonnegative(x: f64) -> Result<f64, ForwardError> {
    finite(x)?;
    if x >= 0.0 {
        Ok(x)
    } else {
        Err(invalid())
    }
}
fn mul(xs: &[f64]) -> Result<f64, ForwardError> {
    for &x in xs {
        nonnegative(x)?;
    }
    if xs.contains(&0.0) {
        return Ok(0.0);
    }
    let mut y = 1.0;
    for &x in xs {
        y *= x;
        if !y.is_normal() {
            return Err(invalid());
        }
    }
    Ok(y)
}
fn divide(x: f64, y: f64) -> Result<f64, ForwardError> {
    finite(x)?;
    if !y.is_normal() || y <= 0.0 {
        return Err(invalid());
    }
    let v = x / y;
    finite(v)?;
    if x != 0.0 && v == 0.0 {
        return Err(invalid());
    }
    Ok(v)
}
/// Immutable point evaluator. Errors return no partially updated state.
/// Admission uses the recovered EOS temperature strictly, not an originally
/// requested constructor temperature. Even sub-ulp endpoint reconstruction
/// outside 1..=1e6 K is rejected; no state/evaluation projection is performed.
/// `h` is nonnegative expansion s^-1; no coordinate conversion is hidden here.
/// Fractions for an absent helium element have exactly zero derivatives.
pub fn igm_point_rhs(
    state: &IgmGasState,
    nh: f64,
    nhe: f64,
    h: f64,
    tcmb: f64,
    photo: IgmPhotoInput,
) -> Result<IgmPointRhs, ForwardError> {
    nonnegative(h)?;
    if !tcmb.is_normal() || tcmb <= 0.0 {
        return Err(invalid());
    }
    for i in 0..3 {
        nonnegative(photo.gamma_s[i])?;
        nonnegative(photo.heat_erg_per_absorber_s[i])?;
        if photo.gamma_s[i] == 0.0 && photo.heat_erg_per_absorber_s[i] != 0.0 {
            return Err(invalid());
        }
    }
    let eos = state.eos(nh, nhe)?;
    let t = eos.temperature_k;
    let ne = eos.electron_density_cm3;
    let rates = igm_rates(t)?;
    let c = HHeModel::controlled_fixture();
    let [x, y, z] = state.fractions;
    let abs = [
        mul(&[nh, 1.0 - x])?,
        mul(&[nhe, 1.0 - (y + z)])?,
        mul(&[nhe, y])?,
    ];
    let ions = [mul(&[nh, x])?, mul(&[nhe, y])?, mul(&[nhe, z])?];
    let mut out = IgmPointRhs::default();
    let mut photo_heat = 0.0;
    let mut photo_bind = 0.0;
    let mut capture_bind = 0.0;
    for i in 0..3 {
        let chi = mul(&[c.threshold_ev[i], c.ev_erg])?;
        out.photo_events_cm3_s[i] = mul(&[abs[i], photo.gamma_s[i]])?;
        out.ci_events_cm3_s[i] = mul(&[ne, abs[i], rates.ci_cm3_s[i]])?;
        out.rr_events_cm3_s[i] = mul(&[ne, ions[i], rates.rr_cm3_s[i]])?;
        out.ci_thermal_sink_erg_cm3_s[i] = mul(&[chi, out.ci_events_cm3_s[i]])?;
        out.rr_thermal_sink_erg_cm3_s[i] = mul(&[ne, ions[i], rates.rr_cooling_erg_cm3_s[i]])?;
        out.ci_floor_events_cm3_s[i] = mul(&[ne, abs[i], rates.diagnostics.ci_floor_cm3_s[i]])?;
        photo_heat += mul(&[abs[i], photo.heat_erg_per_absorber_s[i]])?;
        photo_bind += mul(&[chi, out.photo_events_cm3_s[i]])?;
        capture_bind += mul(&[chi, out.rr_events_cm3_s[i]])?;
    }
    out.dr_events_cm3_s = mul(&[ne, ions[1], rates.dr_cm3_s])?;
    out.dr_thermal_sink_erg_cm3_s = mul(&[ne, ions[1], rates.dr_cooling_erg_cm3_s])?;
    out.excluded_dr_cooling_erg_cm3_s =
        mul(&[ne, ions[1], rates.diagnostics.dr_excluded_erg_cm3_s])?;
    capture_bind += mul(&[c.threshold_ev[1], c.ev_erg, out.dr_events_cm3_s])?;
    let pref = [
        mul(&[ne, abs[0]])?,
        mul(&[ne, ne, ions[1]])?,
        mul(&[ne, ions[1]])?,
    ];
    for (i, &density) in pref.iter().enumerate() {
        out.ce_thermal_sink_erg_cm3_s[i] = mul(&[density, rates.ce_erg[i]])?;
        out.ce_cap_cooling_erg_cm3_s[i] = mul(&[density, rates.diagnostics.ce_cap_erg[i]])?;
    }
    out.freefree_thermal_sink_erg_cm3_s = mul(&[
        ne,
        ions[0] + ions[1] + 4.0 * ions[2],
        rates.freefree_erg_cm3_s,
    ])?;
    let forward: [f64; 3] =
        std::array::from_fn(|i| out.photo_events_cm3_s[i] + out.ci_events_cm3_s[i]);
    let net_h = forward[0] - out.rr_events_cm3_s[0];
    let net_he1 = forward[1] - out.rr_events_cm3_s[1] - out.dr_events_cm3_s;
    let net_he2 = forward[2] - out.rr_events_cm3_s[2];
    let dne = net_h + net_he1 + net_he2;
    out.species_chemical_cm3_s = [-net_h, net_h, -net_he1, net_he1 - net_he2, net_he2, dne];
    out.fraction_dt = [
        divide(net_h, nh)?,
        if nhe == 0.0 {
            0.0
        } else {
            divide(net_he1 - net_he2, nhe)?
        },
        if nhe == 0.0 {
            0.0
        } else {
            divide(net_he2, nhe)?
        },
    ];
    // Nonrelativistic Thomson CMB exchange. Blackbody a_rad derived from exact
    // Planck, Boltzmann and c owners; it does not inherit Grackle's rounded
    // comp_rate normalized to a fixed present-day CMB temperature.
    let a_rad = 8.0 * std::f64::consts::PI.powi(5) * c.kb_erg_k.powi(4)
        / (15.0 * PLANCK_ERG_S.powi(3) * c.c_cm_s.powi(3));
    let cmb_coeff = 4.0 * SIGMA_T_CM2 * a_rad * c.kb_erg_k / (ELECTRON_MASS_G * c.c_cm_s);
    let delta = finite(tcmb - t)?;
    out.cmb_to_gas_erg_cm3_s =
        mul(&[cmb_coeff, ne, tcmb, tcmb, tcmb, tcmb, delta.abs()])? * delta.signum();
    let ci = out.ci_thermal_sink_erg_cm3_s.iter().sum::<f64>();
    let radiative_kin = out.rr_thermal_sink_erg_cm3_s.iter().sum::<f64>()
        + out.dr_thermal_sink_erg_cm3_s
        + out.ce_thermal_sink_erg_cm3_s.iter().sum::<f64>()
        + out.freefree_thermal_sink_erg_cm3_s;
    out.thermal_micro_erg_cm3_s = photo_heat - ci - radiative_kin + out.cmb_to_gas_erg_cm3_s;
    out.binding_micro_erg_cm3_s = photo_bind + ci - capture_bind;
    out.escape_erg_cm3_s = capture_bind + radiative_kin;
    out.photo_input_erg_cm3_s = photo_bind + photo_heat;
    out.expansion_work_erg_cm3_s = mul(&[2.0, h, eos.u_erg_cm3])?;
    out.w_dt_erg_per_h_s = divide(
        out.thermal_micro_erg_cm3_s - out.expansion_work_erg_cm3_s,
        nh,
    )?;
    let particles = finite(nh + nhe + ne)?;
    out.temperature_dt_k_s = divide(
        2.0 * nh * out.w_dt_erg_per_h_s,
        3.0 * c.kb_erg_k * particles,
    )? - divide(t * dne, particles)?;
    for &v in out
        .fraction_dt
        .iter()
        .chain(out.species_chemical_cm3_s.iter())
    {
        finite(v)?;
    }
    for v in [
        photo_heat,
        photo_bind,
        capture_bind,
        ci,
        radiative_kin,
        out.thermal_micro_erg_cm3_s,
        out.binding_micro_erg_cm3_s,
        out.escape_erg_cm3_s,
        out.photo_input_erg_cm3_s,
        out.temperature_dt_k_s,
    ] {
        finite(v)?;
    }
    Ok(out)
}

//! Boundary-valid ideal atomic H/He gas and explicit radiation-unit adapters.
//! No chemistry/cooling provider or physical temperature-fit domain is admitted.
use crate::coupled_primary::PrimaryPacket;
use crate::{
    CharacteristicRay, ForwardError, HHeModel, HHeState, PhotonNode, PhotonPacket, MPC_CM,
};

#[derive(Clone, Copy, Debug)]
pub struct IgmGasState {
    /// HII, HeII, HeIII; neutral fractions are complements, not independent data.
    pub fractions: [f64; 3],
    /// Thermal energy per hydrogen nucleus, erg/H (NOT existing PrimaryState's eV/H).
    pub w_erg_per_h: f64,
}
#[derive(Clone, Copy, Debug)]
pub struct IgmEos {
    pub electron_density_cm3: f64,
    pub temperature_k: f64,
    pub u_erg_cm3: f64,
}
fn invalid() -> ForwardError {
    ForwardError::InvalidInput("IGM_STATE_OR_UNIT_DOMAIN")
}
fn positive(x: f64) -> Result<f64, ForwardError> {
    if x.is_normal() && x > 0.0 {
        Ok(x)
    } else {
        Err(invalid())
    }
}
fn nonnegative(x: f64) -> Result<f64, ForwardError> {
    if x == 0.0 {
        Ok(x)
    } else {
        positive(x)
    }
}
fn product(x: f64, y: f64) -> Result<f64, ForwardError> {
    nonnegative(x)?;
    nonnegative(y)?;
    if x == 0.0 || y == 0.0 {
        Ok(0.0)
    } else {
        positive(x * y)
    }
}
fn ratio(x: f64, y: f64) -> Result<f64, ForwardError> {
    nonnegative(x)?;
    positive(y)?;
    if x == 0.0 {
        Ok(0.0)
    } else {
        positive(x / y)
    }
}
// Reuse the existing owner of kB/eV constants and the algebraic HHe EOS.
// Only validate_state/electron_density/temperature are invoked, never its
// synthetic coefficients, rate evaluation, photon energies or thermal solver.
fn eos_model(n_h: f64, n_he: f64) -> Result<HHeModel, ForwardError> {
    positive(n_h)?;
    nonnegative(n_he)?;
    let mut model = HHeModel::controlled_fixture();
    model.n_h_cm3 = n_h;
    model.n_he_cm3 = n_he;
    model.validate()?;
    Ok(model)
}
fn hhe_state(fractions: [f64; 3], u: f64) -> HHeState {
    HHeState {
        fractions,
        u_erg_cm3: u,
        photon_cm3: [0.0; 3],
        escaped_erg_cm3: 0.0,
    }
}

impl IgmGasState {
    pub fn new(fractions: [f64; 3], w_erg_per_h: f64) -> Result<Self, ForwardError> {
        positive(w_erg_per_h)?;
        eos_model(1.0, 0.0)?.validate_state(&hhe_state(fractions, 0.0))?;
        Ok(Self {
            fractions,
            w_erg_per_h,
        })
    }
    pub fn from_temperature(
        fractions: [f64; 3],
        n_h: f64,
        n_he: f64,
        t: f64,
    ) -> Result<Self, ForwardError> {
        positive(t)?;
        let model = eos_model(n_h, n_he)?;
        let ne = model.electron_density(&hhe_state(fractions, 0.0))?;
        let particles_per_h = positive((n_h + n_he + ne) / n_h)?;
        let w = positive(product(1.5 * model.kb_erg_k, t)? * particles_per_h)?;
        let state = Self::new(fractions, w)?;
        state.eos(n_h, n_he)?;
        Ok(state)
    }
    pub fn eos(&self, n_h: f64, n_he: f64) -> Result<IgmEos, ForwardError> {
        Self::new(self.fractions, self.w_erg_per_h)?;
        let model = eos_model(n_h, n_he)?;
        let u = positive(self.w_erg_per_h * n_h)?;
        let s = hhe_state(self.fractions, u);
        let ne = nonnegative(model.electron_density(&s)?)?;
        // Check positive species contributions so a subnormal product is not
        // silently rounded to zero by the legacy algebraic EOS.
        product(n_h, self.fractions[0])?;
        product(n_he, self.fractions[1] + 2.0 * self.fractions[2])?;
        // The legacy EOS permits subnormal denominators. Our explicit
        // normal-intermediate contract rejects that loss of relative precision.
        positive(3.0 * model.kb_erg_k * positive(n_h + n_he + ne)?)?;
        let t = positive(model.temperature(&s)?)?;
        Ok(IgmEos {
            electron_density_cm3: ne,
            temperature_k: t,
            u_erg_cm3: u,
        })
    }
    /// Exact monatomic adiabatic map at FIXED fractions: w and T scale as a^-2.
    /// No new chemistry evolution, adaptive integration or ledger is implied.
    pub fn adiabatic_to(&self, a0: f64, a1: f64) -> Result<Self, ForwardError> {
        Self::new(self.fractions, self.w_erg_per_h)?;
        let scale = positive(ratio(a0, a1)?)?;
        Self::new(
            self.fractions,
            positive(self.w_erg_per_h * positive(scale * scale)?)?,
        )
    }
}

pub fn w_erg_per_h_to_ev(w: f64) -> Result<f64, ForwardError> {
    ratio(w, HHeModel::controlled_fixture().ev_erg)
}
pub fn w_ev_per_h_to_erg(w: f64) -> Result<f64, ForwardError> {
    product(w, HHeModel::controlled_fixture().ev_erg)
}
fn validate_primary(p: &PrimaryPacket) -> Result<(), ForwardError> {

}
/// Count conversion uses nH0=nH(a)*a^3, NOT the proper endpoint density.
/// Ray occupation is set to zero: packet count is a distinct measure and is

pub fn primary_to_photon_packet(

//! Source-explicit flat radiation + matter + Lambda FLRW background.
//!
//! Coordinates are absolute `ln(a)`, with a=1 today. This deliberately does NOT
//! implement `GeometryBackground::snapshot(t)`: no proper-time inversion is
//! available yet, and passing ln(a) as seconds would be incorrect.
use crate::{ForwardError, GeometrySnapshot};

// NIST CODATA central values, retrieved 2026-10-06:
// G: https://pml.nist.gov/cuu/pdf/RevModPhys.93.025010.pdf (2018 CODATA)
// mp: https://physics.nist.gov/cgi-bin/cuu/Value?mp%7Csearch_for=mass+of+proton
// SI -> cgs: G multiplied by 1e3; proton mass multiplied by 1e3.
const G_CGS: f64 = 6.67430e-8;
const PROTON_MASS_G: f64 = 1.672_621_925_95e-24;

/// All parameters are caller-supplied. `parameter_source` is attribution, not a
/// claim that an external source was verified. No observational default exists.
#[derive(Clone, Debug)]
pub struct FlatFlrwConfig {
    pub h0_per_s: f64,
    pub omega_r: f64,
    pub omega_m: f64,
    pub omega_b: f64,
    pub omega_lambda: f64,
    pub helium_mass_fraction: f64,
    pub tcmb0_k: f64,
    pub ln_a_min: f64,
    pub ln_a_max: f64,
    pub parameter_source: String,
}

#[derive(Clone, Debug)]
pub struct FlatFlrwBackground {
    config: FlatFlrwConfig,
    n_h0_cm3: f64,
    n_he0_cm3: f64,
}

#[derive(Clone, Copy, Debug)]
pub struct FlrwPoint {
    pub a: f64,
    pub redshift: f64,
    pub hubble_per_s: f64,
    pub n_h_cm3: f64,
    pub n_he_cm3: f64,
    pub tcmb_k: f64,
    pub dt_dln_a_s: f64,
    pub dt_dz_s: f64,
    pub geometry: GeometrySnapshot,
}

fn invalid() -> ForwardError {
    ForwardError::InvalidInput("IGM_BACKGROUND_DOMAIN")
}
fn positive(x: f64) -> Result<f64, ForwardError> {
    if x.is_normal() && x > 0.0 {
        Ok(x)
    } else {
        Err(invalid())
    }
}
fn nonnegative(x: f64) -> Result<f64, ForwardError> {
    if x == 0.0 {
        Ok(x)
    } else {
        positive(x)
    }
}
fn product(x: f64, y: f64) -> Result<f64, ForwardError> {
    nonnegative(x)?;
    nonnegative(y)?;
    if x == 0.0 || y == 0.0 {
        Ok(0.0)
    } else {
        positive(x * y)
    }
}

impl FlatFlrwBackground {
    /// Uses rho_crit0=3H0²/(8piG), mH=mp and mHe=4mp. The latter is an
    /// explicit nucleon-mass approximation, not exact atomic masses; binding
    /// and electron-mass corrections are neglected. Omega_r is independently
    /// specified, not inferred from Tcmb (it may include other radiation).
    ///
    /// Flatness tolerance is 1e-12 absolute; inputs are never renormalized.
    /// Positive subnormal intermediates are outside this numerical contract.
    pub fn new(config: FlatFlrwConfig) -> Result<Self, ForwardError> {
        positive(config.h0_per_s)?;
        positive(config.omega_b)?;
        positive(config.tcmb0_k)?;
        for x in [
            config.omega_r,
            config.omega_m,
            config.omega_lambda,
            config.helium_mass_fraction,
        ] {
            nonnegative(x)?;
        }
        if config.omega_b > config.omega_m
            || config.helium_mass_fraction >= 1.0
            || (config.omega_r + config.omega_m + config.omega_lambda - 1.0).abs() > 1e-12
            || !(config.omega_r + config.omega_m + config.omega_lambda).is_finite()
            || !config.ln_a_min.is_finite()
            || !config.ln_a_max.is_finite()
            || config.ln_a_min > config.ln_a_max
            || config.parameter_source.trim().is_empty()
        {
            return Err(invalid());
        }
        let rho_crit = positive(
            3.0 * positive(config.h0_per_s * config.h0_per_s)?
                / (8.0 * std::f64::consts::PI * G_CGS),
        )?;
        let rho_b = positive(config.omega_b * rho_crit)?;
        let n_h0_cm3 =
            positive(positive((1.0 - config.helium_mass_fraction) * rho_b)? / PROTON_MASS_G)?;
        let n_he0_cm3 =
            nonnegative(product(config.helium_mass_fraction, rho_b)? / (4.0 * PROTON_MASS_G))?;
        let bg = Self {
            config,
            n_h0_cm3,
            n_he0_cm3,
        };
        bg.at_ln_a(bg.config.ln_a_min)?;
        bg.at_ln_a(bg.config.ln_a_max)?;
        Ok(bg)
    }

    /// Absolute comoving hydrogen number density, cm^-3 at a=1.
    pub fn n_h_comoving_cm3(&self) -> f64 {
        self.n_h0_cm3
    }

    pub fn at_ln_a(&self, ln_a: f64) -> Result<FlrwPoint, ForwardError> {
        let c = &self.config;
        if !ln_a.is_finite() || ln_a < c.ln_a_min || ln_a > c.ln_a_max {
            return Err(invalid());
        }
        let a = positive(ln_a.exp())?;
        let inv_a = positive(1.0 / a)?;
        let inv_a3 = positive(inv_a.powi(3))?;
        let radiation = if c.omega_r == 0.0 {
            0.0
        } else {
            positive(c.omega_r * positive(inv_a.powi(4))?)?
        };
        let h = positive(
            c.h0_per_s * positive(radiation + product(c.omega_m, inv_a3)? + c.omega_lambda)?.sqrt(),
        )?;
        let redshift = inv_a - 1.0;
        if !redshift.is_finite() || redshift <= -1.0 {
            return Err(invalid());
        }
        let dt_dln_a_s = positive(1.0 / h)?;
        let dt_dz_s = -positive(a / h)?;
        Ok(FlrwPoint {
            a,
            redshift,
            hubble_per_s: h,
            n_h_cm3: positive(self.n_h0_cm3 * inv_a3)?,
            n_he_cm3: product(self.n_he0_cm3, inv_a3)?,
            tcmb_k: positive(c.tcmb0_k * inv_a)?,
            dt_dln_a_s,
            dt_dz_s,
            geometry: GeometrySnapshot::new([a; 3], [h; 3])?,
        })
    }

    pub fn at_redshift(&self, z: f64) -> Result<FlrwPoint, ForwardError> {
        if !z.is_finite() || z <= -1.0 {
            return Err(invalid());
        }
        // Exact representable endpoint identities take precedence over a
        // second transcendental round trip, which can move ln(a) a few ulps
        // beyond its declared bound. No tolerance band or physical clipping:
        // all other inputs pass through the ordinary ln(a) domain checks.
        let early = self.at_ln_a(self.config.ln_a_min)?;
        let late = self.at_ln_a(self.config.ln_a_max)?;
        if z == early.redshift {
            return Ok(early);
        }
        if z == late.redshift {
            return Ok(late);
        }
        self.at_ln_a(-z.ln_1p())
    }
}

    /// Verner ground-state outer-shell fit. Fit thresholds differ from binding energies.
    pub fn cross_section(&self, absorber: Absorber, energy_ev: f64) -> Result<f64, ForwardError> {
        if !energy_ev.is_finite() || energy_ev < 0.0 {
            return Err(ForwardError::InvalidInput("PHOTON_ENERGY_INVALID"));
        }
        if energy_ev > 50000.0 {
            return Err(ForwardError::InvalidInput("VERNER_ENERGY_DOMAIN"));
        }
        let (eth, e0, sigma0, ya, p, yw, y0, y1) = match absorber {
            Absorber::HI => (13.60, 0.4298, 5.475e4, 32.88, 2.963, 0.0, 0.0, 0.0),
            Absorber::HeI => (24.59, 13.61, 949.2, 1.469, 3.188, 2.039, 0.4434, 2.136),
            Absorber::HeII => (54.42, 1.720, 1.369e4, 32.88, 2.963, 0.0, 0.0, 0.0),
        };
        if energy_ev < eth {
            return Ok(0.0);
        }
        let x = energy_ev / e0 - y0;
        let y = (x * x + y1 * y1).sqrt();
        let sigma = sigma0
            * ((x - 1.0).powi(2) + yw * yw)
            * y.powf(p / 2.0 - 5.5)
            * (1.0 + (y / ya).sqrt()).powf(-p)
            * 1e-18;
        if !sigma.is_finite() || sigma < 0.0 {
            return Err(ForwardError::InvalidInput("VERNER_RESULT_INVALID"));
        }
        Ok(sigma)
    }

fn invalid(tag: &'static str) -> ForwardError {
    ForwardError::InvalidInput(tag)
}

fn positive_finite(x: f64, tag: &'static str) -> Result<f64, ForwardError> {
    if x.is_finite() && x > 0.0 {
        Ok(x)
    } else {
        Err(invalid(tag))
    }
}

fn finite(x: f64, tag: &'static str) -> Result<f64, ForwardError> {
    if x.is_finite() {
        Ok(x)
    } else {
        Err(invalid(tag))
    }
}

#[derive(Clone, Copy, Debug, PartialEq)]
pub struct GeometrySnapshot {
    pub scale_factors: [f64; 3],
    pub hubble_per_s: [f64; 3],
}

impl GeometrySnapshot {
    pub fn new(scale_factors: [f64; 3], hubble_per_s: [f64; 3]) -> Result<Self, ForwardError> {
        let snapshot = Self {
            scale_factors,
            hubble_per_s,
        };
        snapshot.validate()?;
        Ok(snapshot)
    }

    fn validate(&self) -> Result<(), ForwardError> {
        for a in self.scale_factors {
            positive_finite(a, "BianchiScale")?;
        }
        for h in self.hubble_per_s {
            finite(h, "BianchiHubble")?;
        }
        self.volume_factor()?;
        Ok(())
    }

    pub fn volume_factor(&self) -> Result<f64, ForwardError> {
        for a in self.scale_factors {
            positive_finite(a, "BianchiScale")?;
        }
        for h in self.hubble_per_s {
            finite(h, "BianchiHubble")?;
        }
        let v = self.scale_factors.iter().product();
        positive_finite(v, "BianchiVolume")
    }
}

/// A prescribed axisymmetric Bianchi-I point in proper-time units.
///
/// The two transverse scale factors are `a exp(-b)` and the longitudinal
/// factor is `a exp(2b)`, so the volume remains `a^3`.  This is a geometry
/// adapter only: it supplies neither a physical source nor gas initial data.
#[derive(Clone, Copy, Debug, PartialEq)]
pub struct AxisymmetricPoint {
    pub mean_scale_factor: f64,
    pub anisotropy: f64,
    pub mean_hubble_per_s: f64,
    pub shear_per_s: f64,
}

impl AxisymmetricPoint {
    pub fn new(
        mean_scale_factor: f64,
        anisotropy: f64,
        mean_hubble_per_s: f64,
        shear_per_s: f64,
    ) -> Result<Self, ForwardError> {
        positive_finite(mean_scale_factor, "AxisymMeanScale")?;
        finite(anisotropy, "AxisymAnisotropy")?;
        finite(mean_hubble_per_s, "AxisymMeanHubble")?;
        finite(shear_per_s, "AxisymShear")?;
        let point = Self {
            mean_scale_factor,
            anisotropy,
            mean_hubble_per_s,
            shear_per_s,
        };
        point.snapshot()?;
        Ok(point)
    }

    pub fn snapshot(&self) -> Result<GeometrySnapshot, ForwardError> {
        let transverse = positive_finite(
            self.mean_scale_factor * (-self.anisotropy).exp(),
            "AxisymTransverseScale",
        )?;
        let longitudinal = positive_finite(
            self.mean_scale_factor * (2.0 * self.anisotropy).exp(),
            "AxisymLongitudinalScale",
        )?;
        GeometrySnapshot::new(
            [transverse, transverse, longitudinal],
            [
                finite(
                    self.mean_hubble_per_s - self.shear_per_s,
                    "AxisymTransverseHubble",
                )?,
                finite(
                    self.mean_hubble_per_s - self.shear_per_s,
                    "AxisymTransverseHubble",
                )?,
                finite(
                    self.mean_hubble_per_s + 2.0 * self.shear_per_s,
                    "AxisymLongitudinalHubble",
                )?,
            ],
        )
    }
}

/// Analytic prescribed axisymmetric geometry over a finite proper-time domain.
///
/// It deliberately has no `ln(a)` inverse: `H=0` is a valid proper-time
/// background, while an inverse chart would be singular there.
#[derive(Clone, Copy, Debug, PartialEq)]
pub struct ConstantAxisymmetricBackground {
    pub reference_time_s: f64,
    pub reference_point: AxisymmetricPoint,
    pub time_domain_s: [f64; 2],
}

impl ConstantAxisymmetricBackground {
    pub fn new(
        reference_time_s: f64,
        reference_point: AxisymmetricPoint,
        time_domain_s: [f64; 2],
    ) -> Result<Self, ForwardError> {
        finite(reference_time_s, "AxisymReferenceTime")?;
        let background = Self {
            reference_time_s,
            reference_point,
            time_domain_s,
        };
        background.validate_time_domain()?;
        Ok(background)
    }

    fn validate_time_domain(&self) -> Result<(), ForwardError> {
        finite(self.time_domain_s[0], "AxisymTimeDomain")?;
        finite(self.time_domain_s[1], "AxisymTimeDomain")?;
        if self.time_domain_s[0] > self.time_domain_s[1] {
            return Err(invalid("AxisymTimeDomain"));
        }
        Ok(())
    }
}

impl GeometryBackground for ConstantAxisymmetricBackground {
    fn snapshot(&self, t: f64) -> Result<GeometrySnapshot, ForwardError> {
        finite(t, "AxisymTime")?;
        self.validate_time_domain()?;
        if t < self.time_domain_s[0] || t > self.time_domain_s[1] {
            return Err(invalid("AxisymTimeDomain"));
        }
        let dt = finite(t - self.reference_time_s, "AxisymTime")?;
        AxisymmetricPoint::new(
            positive_finite(
                self.reference_point.mean_scale_factor
                    * finite(
                        self.reference_point.mean_hubble_per_s * dt,
                        "AxisymExponent",
                    )?
                    .exp(),
                "AxisymMeanScale",
            )?,
            finite(
                self.reference_point.anisotropy
                    + finite(self.reference_point.shear_per_s * dt, "AxisymAnisotropy")?,
                "AxisymAnisotropy",
            )?,
            self.reference_point.mean_hubble_per_s,
            self.reference_point.shear_per_s,
        )?
        .snapshot()
    }
}

impl ConstantHubbleBackground {
    pub fn new(
        initial_scale_factors: [f64; 3],
        hubble_per_s: [f64; 3],
    ) -> Result<Self, ForwardError> {
        GeometrySnapshot::new(initial_scale_factors, hubble_per_s)?;
        Ok(Self {
            initial_scale_factors,
            hubble_per_s,
        })
    }
}

impl GeometryBackground for ConstantHubbleBackground {
    fn snapshot(&self, t: f64) -> Result<GeometrySnapshot, ForwardError> {
        finite(t, "BianchiTime")?;
        GeometrySnapshot::new(self.initial_scale_factors, self.hubble_per_s)?;
        let mut a = [0.0; 3];
        for (i, output) in a.iter_mut().enumerate() {
            let ht = finite(self.hubble_per_s[i] * t, "BianchiExponent")?;
            *output = positive_finite(self.initial_scale_factors[i] * ht.exp(), "BianchiScale")?;
        }
        GeometrySnapshot::new(a, self.hubble_per_s)
    }
}

        g0: &GeometrySnapshot,
        g1: &GeometrySnapshot,
    ) -> Result<([f64; 3], f64), ForwardError> {
        let ray = Self::new(self.energy_ev, self.direction, self.occupation)?;
        g0.validate()?;
        g1.validate()?;
        let mut q = [0.0; 3];
        let mut direct = true;
        for (i, component) in q.iter_mut().enumerate() {
            let ratio = g0.scale_factors[i] / g1.scale_factors[i];
            *component = ray.direction[i] * ratio;
            direct &= ratio.is_finite()
                && ratio > 0.0
                && component.is_finite()
                && (*component != 0.0 || ray.direction[i] == 0.0);
        }
        let r = q[0].hypot(q[1]).hypot(q[2]);
        if direct && r.is_finite() && r > 0.0 {
            let direction = q.map(|value| value / r);
            if direction
                .iter()
                .zip(ray.direction)
                .all(|(a, b)| a.is_finite() && (*a != 0.0 || b == 0.0))
            {
                return Ok((direction, r));
            }
        }
        // A scaled component can underflow while its normalized direction is
        // representable. Rescale before exponentiation rather than losing it.
        let logs: [f64; 3] = std::array::from_fn(|i| {
            if ray.direction[i] == 0.0 {
                f64::NEG_INFINITY
            } else {
                ray.direction[i].abs().ln() + g0.scale_factors[i].ln() - g1.scale_factors[i].ln()
            }
        });
        let largest = logs.iter().copied().fold(f64::NEG_INFINITY, f64::max);
        let scaled = logs.map(|value| (value - largest).exp());
        let norm = scaled[0].hypot(scaled[1]).hypot(scaled[2]);
        let r = positive_finite((largest + norm.ln()).exp(), "RayRedshift")?;
        let mut direction = [0.0; 3];
        for i in 0..3 {
            direction[i] = (logs[i] - largest).exp() / norm * ray.direction[i].signum();
            if !direction[i].is_finite() || (direction[i] == 0.0 && ray.direction[i] != 0.0) {
                return Err(invalid("RayDirectionUnderflow"));
            }
        }
        Ok((direction, r))
    }

    pub fn pullback(
        &self,
        g0: &GeometrySnapshot,
        g1: &GeometrySnapshot,
    ) -> Result<Self, ForwardError> {
        let (direction, r) = self.scaled_components(g0, g1)?;
        let energy = positive_finite(self.energy_ev * r, "RayTransportEnergy")?;
        Self::new(energy, direction, self.occupation)
    }

    pub fn solid_angle_jacobian(
        &self,
        g0: &GeometrySnapshot,
        g1: &GeometrySnapshot,
    ) -> Result<f64, ForwardError> {
        let (_, r) = self.scaled_components(g0, g1)?;
        let log_j = g0
            .scale_factors
            .iter()
            .zip(g1.scale_factors)
            .map(|(a, b)| a.ln() - b.ln())
            .sum::<f64>()
            - 3.0 * r.ln();
        positive_finite(log_j.exp(), "RaySolidAngleJacobian")
    }

    pub fn derivative(&self, g: &GeometrySnapshot) -> Result<RayDerivative, ForwardError> {
        let ray = Self::new(self.energy_ev, self.direction, self.occupation)?;
        g.validate()?;
        let reference = g.hubble_per_s[0];
        let mut shear_projection = 0.0;
        for i in 0..3 {
            shear_projection = finite(
                shear_projection
                    + ray.direction[i] * ray.direction[i] * (g.hubble_per_s[i] - reference),
                "RayHubbleProjection",
            )?;
        }
        let hd = finite(reference + shear_projection, "RayHubbleProjection")?;
        let energy_dot_ev_s = finite(-self.energy_ev * hd, "RayEnergyDerivative")?;
        let mut direction_dot_per_s = [0.0; 3];
        for (i, derivative) in direction_dot_per_s.iter_mut().enumerate() {
            let difference = shear_projection - (g.hubble_per_s[i] - reference);
            *derivative = finite(ray.direction[i] * difference, "RayDirectionDerivative")?;
        }
        Ok(RayDerivative {
            energy_dot_ev_s,
            direction_dot_per_s,
        })
    }
}

            }
            out.collision_per_cm3[a] += other.collision_per_cm3[a];

fn valid_nonnegative(x: f64) -> bool {
    x.is_finite() && x >= 0.0
}

impl HHeModel {
    pub fn controlled_fixture() -> Self {
        Self {
            n_h_cm3: 1e-4,
            n_he_cm3: 8.3e-6,
            c_cm_s: 29_979_245_800.0,
            kb_erg_k: 1.380_649e-16,
            ev_erg: 1.602_176_634e-12,
            threshold_ev: [13.598_434_599_702, 24.587_389_011, 54.417_76],
            photon_energy_ev: [13.7, 24.7, 54.5],
            sigma_cm2: [
                [6e-18, 1e-18, 1e-19],
                [0.0, 7e-18, 1e-18],
                [0.0, 0.0, 1.5e-18],
            ],
            alpha_cm3_s: [2e-13, 1e-12, 3e-13],
            beta_cm3_s: [1e-16, 1e-17, 1e-18],
        }
    }

    pub(crate) fn validate(&self) -> Result<(), ForwardError> {
        if !valid_nonnegative(self.n_h_cm3)
            || !valid_nonnegative(self.n_he_cm3)
            || !(self.n_h_cm3 + self.n_he_cm3).is_finite()
            || self.n_h_cm3 + self.n_he_cm3 <= 0.0
            || !self.c_cm_s.is_finite()
            || self.c_cm_s <= 0.0
            || !self.kb_erg_k.is_finite()
            || self.kb_erg_k <= 0.0
            || !self.ev_erg.is_finite()
            || self.ev_erg <= 0.0
        {
            return Err(ForwardError::InvalidInput("HHE_MODEL_DOMAIN"));
        }
        for a in 0..3 {
            if !valid_nonnegative(self.threshold_ev[a])
                || !valid_nonnegative(self.alpha_cm3_s[a])
                || !valid_nonnegative(self.beta_cm3_s[a])
            {
                return Err(ForwardError::InvalidInput("HHE_MODEL_DOMAIN"));
            }
            for g in 0..3 {
                let sigma = self.sigma_cm2[a][g];
                if !valid_nonnegative(sigma)
                    || (sigma > 0.0 && self.photon_energy_ev[g] < self.threshold_ev[a])
                {
                    return Err(ForwardError::InvalidInput("HHE_SUBTHRESHOLD_EDGE"));
                }
            }
        }
        if self.photon_energy_ev.iter().any(|&x| !valid_nonnegative(x)) {
            return Err(ForwardError::InvalidInput("HHE_MODEL_DOMAIN"));
        }
        Ok(())
    }

    pub(crate) fn validate_state(&self, s: &HHeState) -> Result<(), ForwardError> {
        self.validate()?;
        let [h, he1, he2] = s.fractions;
        if !valid_nonnegative(h)
            || h > 1.0
            || !valid_nonnegative(he1)
            || !valid_nonnegative(he2)
            || he1 + he2 > 1.0
            || !valid_nonnegative(s.u_erg_cm3)
            || !valid_nonnegative(s.escaped_erg_cm3)
            || s.photon_cm3.iter().any(|&x| !valid_nonnegative(x))
        {
            return Err(ForwardError::InvalidInput("HHE_STATE_DOMAIN"));
        }
        Ok(())
    }

    pub fn electron_density(&self, s: &HHeState) -> Result<f64, ForwardError> {
        self.validate_state(s)?;
        let ne =
            self.n_h_cm3 * s.fractions[0] + self.n_he_cm3 * (s.fractions[1] + 2.0 * s.fractions[2]);
        if !ne.is_finite() {
            return Err(ForwardError::InvalidInput("HHE_OVERFLOW"));
        }
        Ok(ne)
    }

    pub fn temperature(&self, s: &HHeState) -> Result<f64, ForwardError> {
        let ne = self.electron_density(s)?;
        let particles = self.n_h_cm3 + self.n_he_cm3 + ne;
        let numerator = 2.0 * s.u_erg_cm3;
        let denominator = 3.0 * self.kb_erg_k * particles;
        if !particles.is_finite()
            || !numerator.is_finite()
            || !denominator.is_finite()
            || denominator <= 0.0
        {
            return Err(ForwardError::InvalidInput("HHE_OVERFLOW"));
        }
        let t = numerator / denominator;
        if !t.is_finite() || (s.u_erg_cm3 > 0.0 && t == 0.0) {
            return Err(ForwardError::InvalidInput("HHE_OVERFLOW"));
        }
        Ok(t)
    }

    pub fn total_energy(&self, s: &HHeState) -> Result<f64, ForwardError> {
        self.validate_state(s)?;
        let [h, he1, he2] = s.fractions;
        let binding = self.ev_erg
            * (self.n_h_cm3 * h * self.threshold_ev[0]
                + self.n_he_cm3
                    * (he1 * self.threshold_ev[1]
                        + he2 * (self.threshold_ev[1] + self.threshold_ev[2])));
        let photons = self.ev_erg
            * (0..3)
                .map(|g| self.photon_energy_ev[g] * s.photon_cm3[g])
                .sum::<f64>();
        let total = s.u_erg_cm3 + binding + photons + s.escaped_erg_cm3;
        if !total.is_finite() {
            return Err(ForwardError::InvalidInput("HHE_OVERFLOW"));
        }
        Ok(total)
    }
}

impl HHeState {
    pub fn controlled_fixture(model: &HHeModel) -> Self {
        let fractions = [0.01, 0.019, 0.001];
        let ne =
            model.n_h_cm3 * fractions[0] + model.n_he_cm3 * (fractions[1] + 2.0 * fractions[2]);
        Self {

        }
    }

            self.photon_cm3[1],
            self.photon_cm3[2],
        ]
    }
}

pub fn hhe_rhs(model: &HHeModel, state: &HHeState) -> Result<HHeRhs, ForwardError> {
    model.validate_state(state)?;
    let [h, he1, he2] = state.fractions;
    let lower = [
        model.n_h_cm3 * (1.0 - h),
        model.n_he_cm3 * (1.0 - (he1 + he2)),
        model.n_he_cm3 * he1,
    ];
    let upper = [
        model.n_h_cm3 * h,
        model.n_he_cm3 * he1,
        model.n_he_cm3 * he2,
    ];
    let ne = model.electron_density(state)?;
    let temperature = model.temperature(state)?;
    let mut photo = [[0.0; 3]; 3];
    let mut collision = [0.0; 3];
    let mut recombination = [0.0; 3];
    let mut j = [0.0; 3];
    let mut du = 0.0;
    let mut escape = 0.0;
    let mut photon_dot = [0.0; 3];
    for a in 0..3 {
        collision[a] = lower[a] * ne * model.beta_cm3_s[a];
        recombination[a] = upper[a] * ne * model.alpha_cm3_s[a];
        j[a] = collision[a] - recombination[a];
        du -= collision[a] * model.threshold_ev[a] * model.ev_erg
            + recombination[a] * 1.5 * model.kb_erg_k * temperature;
        escape += recombination[a]
            * (model.threshold_ev[a] * model.ev_erg + 1.5 * model.kb_erg_k * temperature);
        for g in 0..3 {
            photo[a][g] = model.c_cm_s * lower[a] * model.sigma_cm2[a][g] * state.photon_cm3[g];
            j[a] += photo[a][g];
            photon_dot[g] -= photo[a][g];
            du += photo[a][g] * (model.photon_energy_ev[g] - model.threshold_ev[a]) * model.ev_erg;
        }
    }
    // Per-capita equations remain meaningful if one nuclear density is zero.
    let per_capita = |a: usize| {
        let lower_fraction = [1.0 - h, 1.0 - (he1 + he2), he1][a];
        let upper_fraction = [h, he1, he2][a];
        let photo_rate: f64 = (0..3)
            .map(|g| model.c_cm_s * model.sigma_cm2[a][g] * state.photon_cm3[g])
            .sum();
        lower_fraction * (photo_rate + ne * model.beta_cm3_s[a])
            - upper_fraction * ne * model.alpha_cm3_s[a]
    };
    let d = [
        per_capita(0),
        per_capita(1) - per_capita(2),
        per_capita(2),
        du,
        photon_dot[0],
        photon_dot[1],
        photon_dot[2],
    ];
    if d.iter().any(|x| !x.is_finite())
        || !escape.is_finite()
        || photo.iter().flatten().any(|x| !x.is_finite())
        || collision
            .iter()
            .chain(recombination.iter())
            .any(|x| !x.is_finite())
    {
        return Err(ForwardError::InvalidInput("HHE_OVERFLOW"));
    }
    Ok(HHeRhs {
        derivative: d,
        photo_per_cm3_s: photo,
        collision_per_cm3_s: collision,
        recombination_per_cm3_s: recombination,
        escaped_energy_rate: escape,
    })
}

//! Read a frozen key=value config and run an explicitly bound local case.
use std::{fs, path::PathBuf};
fn main() {
    let args = std::env::args().skip(1).collect::<Vec<_>>();
    let mut config = None;
    let mut output = None;
    let mut case = None;
    let mut resume = false;
    let mut i = 0;
    while i < args.len() {
        match args[i].as_str() {
            "--resume" => resume = true,
            "--config" | "--output" | "--case" => {
                let key = &args[i];
                i += 1;
                let value = args
                    .get(i)
                    .unwrap_or_else(|| {
                        eprintln!("missing {key} value");
                        std::process::exit(2)
                    })
                    .clone();
                match key.as_str() {
                    "--config" => config = Some(value),
                    "--output" => output = Some(value),
                    _ => case = Some(value),
                }
            }
            _ => {
                eprintln!("unknown argument {}", args[i]);
                std::process::exit(2)
            }
        }
        i += 1;
    }
    let (Some(config), Some(output), Some(case)) = (config, output, case) else {
        eprintln!("usage: axbi_reionization --config PATH --output NEW_DIR --case ID [--resume]");
        std::process::exit(2)
    };
    let result = fs::read_to_string(&config)
        .map_err(|e| format!("{e:?}"))
        .and_then(|text| {
            rei_microphysics::axbi_driver::run(&text, &PathBuf::from(output), &case, resume)
                .map_err(|e| format!("{e:?}"))
        });
    if let Err(e) = result {
        eprintln!("{e}");
        std::process::exit(1)
    }
}
