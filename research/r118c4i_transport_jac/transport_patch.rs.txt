    }

    }
    pub(crate) fn c4i_tangent(&self,a:[f64;3],nh:f64,dt:f64,scheme:u8)->[[f64;3];13]{
        let model=HHeModel::controlled_fixture();let mut out=[[0.;3];13];let mut corr=[[0.;3];13];let cd=model.c_cm_s*dt;
        for n in &self.nodes{
            let x=cd*sum((0..3).map(|i|a[i]*n.sigma[i]));let (e,phi,psi)=exponential_factors(x);
            let (dp,dpsi)=if x<0.05{
                let mut f=0.;let mut g=0.;let mut p=1.;let mut fact=1.;
                for k in 1..=14{fact*= (k+1) as f64;let sign=if k%2==0{1.}else{-1.};f+=sign*k as f64*p/fact;g+=sign*k as f64*p/(fact*(k+2) as f64);p*=x;}(f,g)
            }else{let dp=(e-phi)/x;(dp,(-dp-psi)/x)};
            let (ds,mean,dm)=match scheme{0=>{let p=1./(1.+x);(-p*p,p,-p*p)},1=>(-e,phi,dp),_=>(dp,psi,dpsi)};
            for j in 0..3{

                for i in 0..13{compensated_add(&mut out[i][j],&mut corr[i][j],v[i]);}
            }
        }out
    }
