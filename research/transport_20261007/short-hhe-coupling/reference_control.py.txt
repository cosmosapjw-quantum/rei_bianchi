"""Bounded invocation of the existing independent continuous-source Radau lane.
Physical cfg remains immutable. Only the authorized end coordinate is shortened.
This is a different continuous-time discretization, never a pure-time BE claim.
"""
import sys
sys.dont_write_bytecode=True
import csv,hashlib,json,math,pathlib,signal,time
D=pathlib.Path(__file__).resolve().parent;R=D.parent.parent
sys.path.insert(0,str(R/'baseline-repository/tools'))
import numpy as np
import igm_reference as physics
import igm_continuous_reference as reference
import igm_compare as compare
level=sys.argv[1];rtol,atol={'tight':(1e-10,1e-13),'tighter':(1e-11,1e-14)}[level]
config=physics.read_config(R/'archived-research/long-flrw/configs/igm_manufactured_z12_to10.cfg')
s0=-math.log1p(config['z_start']);s1=s0+config['max_dln_a'];config['z_end']=math.expm1(-s1)
assert -math.log1p(config['z_end'])==s1
# Independent closed-form positive Gauss4 on the identical declared edge union.
lo=s0+math.log(config['energy_min_ev']);hi=s1+math.log(config['energy_max_ev'])
bounds=[lo+(hi-lo)*i/512 for i in range(513)]
for i in range(25):
 s=s0+config['max_dln_a']*i/24
 for e in [config['energy_min_ev'],config['energy_max_ev'],*physics.CUTOFF]:
  v=s+math.log(e)
  if lo<v<hi:bounds.append(v)
bounds=sorted(set(bounds))
a=math.sqrt((3-2*math.sqrt(6/5))/7);b=math.sqrt((3+2*math.sqrt(6/5))/7)
wi=(18+math.sqrt(30))/36;wo=(18-math.sqrt(30))/36
nodes=[];weights=[]
for l,r in zip(bounds[:-1],bounds[1:]):
 for x,w in [(-b,wo),(-a,wi),(a,wi),(b,wo)]:
  eta=l+(r-l)*.5*(1+x)
  assert l<eta<r
  nodes.append(eta);weights.append((r-l)*.5*w)
assert len(nodes)<=4096
eta=np.array(nodes);weights=np.array(weights)
# The imported numerical implementation remains byte-for-byte unchanged.
reference.spectral_grid=lambda *args,**kwargs:(eta,weights)
times=np.array([s0+config['max_dln_a']*i/3 for i in range(4)])
times[0]=s0;times[-1]=s1
status={'method':'independent existing continuous-source Radau, same fixed eta spectrum','label':level,'rtol':rtol,'atol':atol,'unique_live_spectral_abscissae':len(nodes),'live_sample_interpretation':'distinct spectral quadrature abscissae; solver vectors reuse the same node set','output_times':times.tolist(),'budget_seconds_per_attempt':45,'completed':False}
# Each attempt is subordinate to the shared frozen 120 CPU / 180 wall supervisor.
def timeout(*_):raise TimeoutError('45-second independent-reference feasibility bound')
signal.signal(signal.SIGALRM,timeout);signal.alarm(45)
start=time.monotonic()
try:
 result=reference.solve_history(config,times,spectral_panels=512,order=4,rtol=rtol,atol=atol,method='Radau')
 signal.alarm(0)
 with (D/f'results/reference_{level}.csv').open('w') as f:
  writer=csv.DictWriter(f,fieldnames=list(result['rows'][0]));writer.writeheader();writer.writerows(result['rows'])
 status.update(completed=True,stats=result['stats'],comparison_with_B_m8_g4=compare.compare_rows(compare.read_csv(D/'results/B_phases_m8_g4.csv'),result['rows']))
 if level=='tighter':
  if (D/'results/reference_tight.csv').exists():status['reference_retightening']=compare.compare_rows(compare.read_csv(D/'results/reference_tight.csv'),result['rows'],allowance_factor=.1)
  else:status['reference_retightening']='UNVERIFIED: preceding tight attempt rejected its original accepted-state number budget'
except Exception as e:status['failure']=repr(e)
finally:
 status['wall_seconds']=time.monotonic()-start
 (D/f'results/REFERENCE_{level.upper()}_STATUS.json').write_text(json.dumps(status,indent=2)+'\n')
print(json.dumps({k:v for k,v in status.items() if k not in ['comparison_with_B_m8_g4','reference_retightening','stats']}))
raise SystemExit(0 if status['completed'] else 2)
